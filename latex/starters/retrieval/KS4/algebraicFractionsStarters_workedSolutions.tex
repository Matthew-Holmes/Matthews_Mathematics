% Retrieval starters for a Year 10 mixed higher/foundation class,
% used across the algebraic fractions unit.
%
% Every starter draws on each of the three earlier topics: prime factor
% decomposition (factor trees, and HCF and LCM from Venn diagrams),
% indices (the index laws, negative and fractional indices) and surds
% (simplifying, adding, subtracting and multiplying). Each number skill
% is set beside the algebra it becomes once the fractions have letters
% in them. The five starters follow the order of the unit:
%
%   1  HCF and cancelling           -> simplifying single-term fractions
%   2  index laws                   -> dividing powers, negative indices
%   3  surds and single brackets    -> simplifying by factorising
%   4  double brackets              -> simplifying by factorising a quadratic
%   5  LCM and fraction arithmetic  -> adding, subtracting, multiplying
%                                      and dividing
%
% Within each starter the questions run from foundation to higher. A
% question marked (Extension) is higher tier, and the last question on
% each starter is a challenge that connects to the lesson.
% ================================================================

\begin{frame}[t]{Worked Solution: Starter 1, Question 1}
\hypertarget{work-st1}{}
\textbf{Question:} Simplify $\sqrt{20}$.
\awork{Solution: \(20=4\times5\), and \(4\) is a square number. \par
So \(\sqrt{20}=\sqrt{4\times5}=\sqrt{4}\times\sqrt{5}=2\sqrt{5}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 2}
\textbf{Question:} Work out $2^{-1}$ and $9^{\frac12}$.
\awork{Solution: A negative index means the reciprocal: \(2^{-1}=\frac{1}{2}\). \par
A power of \(\frac12\) means the square root: \(9^{\frac12}=\sqrt{9}=3\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 3}
\textbf{Question:} Complete the factor tree. Write $84$ as a product of prime factors in index form.
\awork{Solution: Split \(84=2\times42\). Then \(42=2\times21\), and \(21=3\times7\). \par
So \(84=2\times2\times3\times7=2^2\times3\times7\). \par
In the tree, the missing box is \(21\), which splits into the primes \(3\) and \(7\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 4}
\textbf{Question:} $60=2^2\times3\times5$. Complete Venn diagram A for $60$ and $84$, then find the HCF and the LCM of $60$ and $84$.
\awork{Solution: Write \(84=2^2\times3\times7\). \par
The shared prime factors are \(2^2\) and \(3\), so the HCF is \(2^2\times3=12\). \par
The LCM uses every prime factor to its highest power: \(2^2\times3\times5\times7=420\). \par
In Venn diagram A, put \(2,2,3\) in the intersection, \(5\) only on the \(60\) side, and \(7\) only on the \(84\) side.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 5}
\textbf{Question:} Simplify $\dfrac{60}{84}$ by dividing by the HCF. Which numbers in Venn diagram A are not in the HCF?
\awork{Solution: From Question 4, the HCF is \(12\). \par
\(\frac{60}{84}=\frac{60\div12}{84\div12}=\frac{5}{7}\). \par
The numbers not in the HCF are \(5\) (only in \(60\)) and \(7\) (only in \(84\)); these are the numbers left in the simplified fraction.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 6}
\textbf{Question:} Simplify $\dfrac{x^5}{x^2}$ by writing out the $x$s and cancelling.
\awork{Solution: \(\frac{x^5}{x^2}=\frac{x\times x\times x\times x\times x}{x\times x}\). \par
Cancel two \(x\)s from the top and bottom: \(\frac{x\times x\times x}{1}=x^3\). \par
By index laws, this is \(x^{5-2}=x^3\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 7}
\textbf{Question:} (Challenge) Complete Venn diagram B for $6a^2b$ and $9ab^2$. Use it to simplify $\dfrac{6a^2b}{9ab^2}$.
\awork{Solution: \(6a^2b=2\times3\times a\times a\times b\). \par
\(9ab^2=3\times3\times a\times b\times b\). \par
The shared factors are \(3\), \(a\) and \(b\), so the HCF is \(3ab\). \par
The remaining factors are \(2a\) on the \(6a^2b\) side and \(3b\) on the \(9ab^2\) side. \par
So \(\frac{6a^2b}{9ab^2}=\frac{3ab\times2a}{3ab\times3b}=\frac{2a}{3b}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 1}
\hypertarget{work-st2}{}
\textbf{Question:} Write $72$ as a product of prime factors in index form.
\awork{Solution: \(72=8\times9\). \par
\(8=2^3\) and \(9=3^2\). \par
So \(72=2^3\times3^2\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 2}
\textbf{Question:} Simplify $\sqrt{8}+\sqrt{18}$.
\awork{Solution: \(\sqrt{8}=\sqrt{4\times2}=2\sqrt{2}\). \par
\(\sqrt{18}=\sqrt{9\times2}=3\sqrt{2}\). \par
So \(\sqrt{8}+\sqrt{18}=2\sqrt{2}+3\sqrt{2}=5\sqrt{2}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 3}
\textbf{Question:} Simplify $x^4\times x^3$ and $y^8\div y^2$.
\awork{Solution: For multiplication, add the indices: \(x^4\times x^3=x^{4+3}=x^7\). \par
For division, subtract the indices: \(y^8\div y^2=y^{8-2}=y^6\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 4}
\textbf{Question:} Write $2^{-3}$ and $x^{-2}$ as fractions.
\awork{Solution: A negative index means take the reciprocal. \par
\(2^{-3}=\frac{1}{2^3}=\frac{1}{8}\). \par
\(x^{-2}=\frac{1}{x^2}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 5}
\textbf{Question:} Find the volume of the cube. Is it $2x^3$ or $8x^3$?
\awork{Solution: The cube has side length \(2x\). \par
Volume \(=(2x)^3=2^3\times x^3=8x^3\). \par
So it is \(8x^3\), not \(2x^3\); the \(2\) must also be cubed.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 6}
\textbf{Question:} Find the volume of the cuboid.
\awork{Solution: The cuboid has dimensions \(3x\), \(x^2\) and \(2x\). \par
Volume \(=3x\times x^2\times2x\). \par
Multiply the numbers: \(3\times2=6\). \par
Add the indices of \(x\): \(x^1\times x^2\times x^1=x^{1+2+1}=x^4\). \par
So the volume is \(6x^4\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 7}
\textbf{Question:} The square has area $36x^4$, so its side length is $\left(36x^4\right)^{\frac12}$. Simplify this.
\awork{Solution: \(\left(36x^4\right)^{\frac12}=36^{\frac12}\times\left(x^4\right)^{\frac12}\). \par
\(36^{\frac12}=\sqrt{36}=6\). \par
\(\left(x^4\right)^{\frac12}=x^{4\times\frac12}=x^2\). \par
So the side length is \(6x^2\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 8}
\textbf{Question:} (Extension) Complete the working: \(8^{-\frac23}=\frac{1}{8^{\frac23}}=\frac{1}{(\sqrt[3]{8})^2}=\frac{1}{\Box^2}=\Box\).
\awork{Solution: \(\sqrt[3]{8}=2\) because \(2^3=8\). \par
So \(8^{\frac23}=\left(\sqrt[3]{8}\right)^2=2^2=4\). \par
Therefore \(8^{-\frac23}=\frac{1}{4}\). \par
The blanks are \(2\) and \(\frac14\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 9}
\textbf{Question:} (Challenge) Divide the volume of the cube by the volume of the cuboid. Write the answer as a fraction in its simplest form.
\awork{Solution: From Questions 5 and 6, the cube has volume \(8x^3\) and the cuboid has volume \(6x^4\). \par
So the quotient is \(\frac{8x^3}{6x^4}=\frac{8}{6}\times\frac{x^3}{x^4}=\frac{4}{3}\times\frac{1}{x}=\frac{4}{3x}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 4, Question 2}
\textbf{Question:} Simplify $\sqrt{50}-\sqrt{8}$. Is it $\sqrt{42}$?

So \(\sqrt{50}-\sqrt{8}=5\sqrt{2}-2\sqrt{2}=3\sqrt{2}\). \par

\begin{frame}[t]{Worked Solution: Starter 4, Question 3}

\begin{frame}[t]{Worked Solution: Starter 5, Question 3}
\textbf{Question:} $12=2^2\times3$ and $18=2\times3^2$. Complete Venn diagram A, then find the HCF and the LCM of $12$ and $18$.

\(\frac{7}{18}=\frac{7\times2}{18\times2}=\frac{14}{36}\). \par
So \(\frac{5}{12}+\frac{7}{18}=\frac{15}{36}+\frac{14}{36}=\frac{29}{36}\). \par
The blanks are \(15\), \(14\) and \(29\).}

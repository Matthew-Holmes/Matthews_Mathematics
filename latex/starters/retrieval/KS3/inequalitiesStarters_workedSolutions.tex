% unit.
%
% Every starter draws on each of the three earlier topics: fractions,
% decimals and percentages; probability, including frequency trees; and
% standard form, including calculating with it. Each starter uses more of
% the inequalities unit than the one before, in the order it is taught:
%
%   1  comparing and ordering numbers with < and >, and reading x > 0.3
%   2  one-sided inequalities on a number line: open and filled circles,
%      and the integers that satisfy them
%   3  double inequalities. The rule for standard form, 1 <= A < 10, and
%      the probability scale, 0 <= p <= 1, are both double inequalities.
%   4  solving one- and two-step inequalities, and forming one from a
%      frequency tree
%   5  unknowns on both sides, solving double inequalities, and forming
%      inequalities from probabilities
%
% Within each starter the questions run from retrieval of the earlier
% topics to the new idea, and the last question on each starter is a
% challenge that connects to the lesson.

\begin{frame}[t]{Worked Solution: Starter 1, Question 1}
\hypertarget{work-st1}{}
\footnotesize
\textbf{Question:} Write $\frac38$ as a decimal and as a percentage.

\awork{Solution: \(\frac38 = 3 \div 8 = 0.375\). To get a percentage, multiply by \(100\): \(0.375 \times 100 = 37.5\%\). So \(\frac38 = 0.375 = 37.5\%\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 2}
\footnotesize
\textbf{Question:} Write $32\,000$ in standard form.

\awork{Solution: Standard form is \(A\times10^n\) with \(1\le A<10\). Write \(32\,000 = 3.2 \times 10\,000 = 3.2\times10^4\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 3}
\footnotesize
\textbf{Question:} Mark $32\%$, $0.35$ and $\frac38$ on number line A. Write them in order, using $<$.

\awork{Solution: Convert to decimals: \(32\%=0.32\), \(0.35=0.35\), \(\frac38=0.375\). On the number line from \(0.3\) to \(0.4\), \(0.32\) is \(2\) hundredths along, \(0.35\) is \(5\) hundredths along, and \(0.375\) is \(7.5\) hundredths along. So the order is \(32\% < 0.35 < \frac38\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 4}
\footnotesize
\textbf{Question:} Write $<$ or $>$ between \(3.2\times10^4\) and \(9.8\times10^3\), and between \(6\times10^{-3}\) and \(2\times10^{-2}\).

\awork{Solution: \(3.2\times10^4=32\,000\) and \(9.8\times10^3=9\,800\), so \(3.2\times10^4 > 9.8\times10^3\). For the second, \(6\times10^{-3}=0.006\) and \(2\times10^{-2}=0.02\), so \(6\times10^{-3} < 2\times10^{-2}\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 5}
\footnotesize
\textbf{Question:} $60$ people watched a film. Complete the frequency tree.

\awork{Solution: The total is \(60\). The adult branch is \(35\), so the child branch is \(60-35=25\). For adults, \(21\) bought popcorn, so \(35-21=14\) did not. For children, \(18\) bought popcorn, so \(25-18=7\) did not. Thus the missing values are: child \(25\), adult no popcorn \(14\), child no popcorn \(7\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 6}
\footnotesize
\textbf{Question:} An adult is chosen at random. Find the probability that they bought popcorn, as a percentage. Do the same for a child. Write $<$ or $>$ between them.

\awork{Solution: For adults: \(P(\text{popcorn})=\frac{21}{35}=0.6=60\%\). For children: \(P(\text{popcorn})=\frac{18}{25}=0.72=72\%\). Since \(60\% < 72\%\), the adult probability is less than the child probability.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 7}
\footnotesize
\textbf{Question:} $x>0.3$ means ``$x$ is greater than $0.3$''. Which of these could $x$ be? \(\frac25\), \(0.29\), \(3\times10^{-1}\), \(31\%\), \(\frac14\).

\awork{Solution: Convert each to a decimal. \(\frac25=0.4\), \(0.29=0.29\), \(3\times10^{-1}=0.3\), \(31\%=0.31\), \(\frac14=0.25\). We need \(x>0.3\), so \(x\) must be strictly more than \(0.3\). The valid values are \(\frac25=0.4\) and \(31\%=0.31\). \(3\times10^{-1}=0.3\) is not valid because it is equal to \(0.3\), not greater.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 1, Question 8}
\footnotesize
\textbf{Question:} (Challenge) $0.2>0.15$. Is $-0.2>-0.15$? Explain using a number line.

\awork{Solution: No. On a number line, negative numbers are to the left of \(0\). The point \(-0.2\) is further left than \(-0.15\), so \(-0.2 < -0.15\). The order reverses when taking negatives: from \(0.2>0.15\), we get \(-0.2<-0.15\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 1}
\hypertarget{work-st2}{}
\footnotesize
\textbf{Question:} Write $0.45$ as a fraction and as a percentage.

\awork{Solution: \(0.45=\frac{45}{100}=\frac{9}{20}\). Multiply by \(100\): \(0.45\times100=45\%\). So \(0.45=\frac{9}{20}=45\%\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 2}
\footnotesize
\textbf{Question:} Write $6.1\times10^{-3}$ as an ordinary number.

\awork{Solution: \(10^{-3}\) means divide by \(1000\), so \(6.1\times10^{-3}=6.1\div1000=0.0061\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 3}
\footnotesize
\textbf{Question:} Draw $x\geq2$ on number line A. Can $x$ be equal to $2$?

\awork{Solution: The inequality \(x\ge2\) means \(x\) is \(2\) or greater. On number line A, place a filled circle at \(2\) and draw an arrow to the right. Yes, \(x\) can equal \(2\), because the \(\ge\) sign includes equality.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 4}
\footnotesize
\textbf{Question:} Write the inequality on number line B, and two integers that satisfy it.

\awork{Solution: Number line B has an open circle at \(-1\) and an arrow pointing left. The open circle means \(-1\) is not included, so the inequality is \(x<-1\). Two integers that satisfy it are \(-2\) and \(-3\) (any integer less than \(-1\) works).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 5}
\footnotesize
\textbf{Question:} Draw $y>\frac32$ on number line C. Does $y=150\%$ satisfy it?

\awork{Solution: \(\frac32=1.5\). On number line C, draw an open circle at \(1.5\) and an arrow to the right. \(150\%=1.5\), which equals \(\frac32\), but the inequality is \(y>\frac32\), so \(y\) must be strictly greater. Therefore \(y=150\%\) does not satisfy it.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 6}
\footnotesize
\textbf{Question:} Which of these satisfy $n>4\times10^3$? \(4000\), \(3.9\times10^4\), \(0.5\times10^4\), \(4.01\times10^3\).

\awork{Solution: \(4\times10^3=4000\). Compare each: \(4000\) is equal to \(4000\), so it does not satisfy \(n>4000\). \(3.9\times10^4=39\,000>4000\), so yes. \(0.5\times10^4=5000>4000\), so yes. \(4.01\times10^3=4010>4000\), so yes. Therefore all except \(4000\) satisfy the inequality.}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 7}
\footnotesize
\textbf{Question:} A match ends in a win, a draw or a loss. $P(\text{win})=0.35$ and $P(\text{draw})=\frac15$. Find $P(\text{loss})$ as a percentage. Write $<$ or $>$ between $P(\text{win})$ and $P(\text{loss})$.

\awork{Solution: Convert \(P(\text{draw})=\frac15=0.2\). The probabilities sum to \(1\), so \(P(\text{loss})=1-0.35-0.2=0.45=45\%\). Compare \(P(\text{win})=35\%\) and \(P(\text{loss})=45\%\), so \(P(\text{win})<P(\text{loss})\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 2, Question 8}
\footnotesize
\textbf{Question:} (Challenge) $x$ is an integer, $x>-3$ and $x\leq2$. List the values $x$ can take.

\awork{Solution: The inequality \(x>-3\) means \(x\) is greater than \(-3\), so \(-3\) is not included. The inequality \(x\le2\) means \(x\) is less than or equal to \(2\), so \(2\) is included. The integers satisfying both are \(-2,\ -1,\ 0,\ 1,\ 2\).}
\end{frame}

\awork{Solution: \(\frac{7}{20}=\frac{7\times5}{20\times5}=\frac{35}{100}=0.35\). As a percentage, \(0.35\times100=35\%\). So \(\frac{7}{20}=0.35=35\%\).}

\begin{frame}[t]{Worked Solution: Starter 3, Question 2}

\textbf{Question:} Work out $(3\times10^4)\times(2\times10^3)$.

\awork{Solution: Multiply the numbers and the powers of \(10\) separately: \(3\times2=6\) and \(10^4\times10^3=10^{4+3}=10^7\). So the product is \(6\times10^7\).}

\begin{frame}[t]{Worked Solution: Starter 3, Question 3}

\begin{frame}[t]{Worked Solution: Starter 3, Question 4}
\footnotesize
\textbf{Question:} Standard form is $A\times10^n$. Number line B shows the values $A$ can take. Write them as a double inequality.

\awork{Solution: Number line B runs from a filled circle at \(1\) to an open circle at \(10\). This means \(A\) is at least \(1\) but strictly less than \(10\), so the double inequality is \(1\le A<10\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 3, Question 5}
\footnotesize
\textbf{Question:} Which of these are in standard form? \(0.8\times10^3\), \(10\times10^2\), \(1\times10^5\), \(9.9\times10^{-4}\).

\awork{Solution: Standard form requires \(A\) to satisfy \(1\le A<10\). \(0.8\times10^3\) is not standard form because \(0.8<1\). \(10\times10^2\) is not standard form because \(10\) is not less than \(10\). \(1\times10^5\) is standard form. \(9.9\times10^{-4}\) is standard form. \awork{Solution: \(\frac14=0.25\) and \(40\%=0.4\), so the inequality becomes \(0.25<x<0.4\). Now \(\frac13=0.\overline{3}\approx0.333\), which lies between \(0.25\) and \(0.4\). Yes, \(x=\frac13\) satisfies it.}

\begin{frame}[t]{Worked Solution: Starter 3, Question 7}

\end{frame}

\begin{frame}[t]{Worked Solution: Starter 3, Question 8}
\footnotesize
\textbf{Question:} (Challenge) $0.6<P(\text{rain})\leq1$. Write a double inequality for $P(\text{no rain})$.

\awork{Solution: Since \(P(\text{no rain})=1-P(\text{rain})\), the direction of the inequalities reverses. If \(P(\text{rain})>0.6\), then \(P(\text{no rain})<1-0.6=0.4\). If \(P(\text{rain})\le1\), then \(P(\text{no rain})\ge1-1=0\). So \(0\le P(\text{no rain})<0.4\).}
\end{frame}

\begin{frame}[t]{Worked Solution: Starter 5, Question 3}

\awork{Solution: Subtract \(2x\) from both sides: \(3x-3\le9\). Add \(3\) to both sides: \(3x\le12\). Divide both sides by \(3\): \(x\le4\). So the solution is \(x\le4\).}

\begin{frame}[t]{Worked Solution: Starter 5, Question 4}

\textbf{Question:} A company made $5\times10^3$ phones. Complete the frequency tree.

% ---------- Worked solutions: Starter 1 ----------

\begin{frame}[label=starter1sol]{Starter 1 Q1}
\textbf{Question:} Calculate $-5+12$.

\textbf{Worked solution:} Start at $-5$ on a number line and move $12$ units to the right. This lands at $7$, so $-5+12=7$.

\textbf{Answer:} $7$.
\end{frame}

\begin{frame}{Starter 1 Q2}
\textbf{Question:} Collect like terms: $3x+4x$.

\textbf{Worked solution:} Both terms have the same variable part $x$, so add the coefficients:
\[
3x+4x=(3+4)x=7x.
\]

\textbf{Answer:} $7x$.
\end{frame}

\begin{frame}{Starter 1 Q3}
\textbf{Question:} Expand $3(x+4)$.

\textbf{Worked solution:} Multiply each term inside the brackets by $3$:
\[
3(x+4)=3\cdot x+3\cdot 4=3x+12.
\]

\textbf{Answer:} $3x+12$.
\end{frame}

\begin{frame}{Starter 1 Q4}
\textbf{Question:} Solve $x+6=14$.

\textbf{Worked solution:} Subtract $6$ from both sides:
\[
x=14-6=8.
\]

Check: $8+6=14$.

\textbf{Answer:} $x=8$.
\end{frame}

\begin{frame}{Starter 1 Q5}
\textbf{Question:} The bar model has two $x$-blocks and a $4$-block, with total $12$. Write the equation and solve it.

\textbf{Worked solution:} The diagram shows
\[
2x+4=12.
\]
Subtract $4$ from both sides:
\[
2x=8.
\]
Divide both sides by $2$:
\[
x=4.
\]

\textbf{Answer:} $x=4$.
\end{frame}

\begin{frame}{Starter 1 Q6}
\textbf{Question:} If the total was $18$ instead of $12$, what would $x$ be?

\textbf{Worked solution:} The new equation is
\[
2x+4=18.
\]
Subtract $4$:
\[
2x=14.
\]
Divide by $2$:
\[
x=7.
\]

\textbf{Answer:} $x=7$.
\end{frame}

\begin{frame}{Starter 1 Q7}
\textbf{Question:} Is $3(x+4)=3x+4$ true or false? Explain.

\textbf{Worked solution:} Expand the left-hand side:
\[
3(x+4)=3x+12.
\]
This is not equal to $3x+4$, because $3x+12\neq 3x+4$ (subtracting $3x$ gives $12\neq 4$).

\textbf{Answer:} False, because $3(x+4)=3x+12$.
\end{frame}

\begin{frame}{Starter 4 Q7}
\textbf{Question:} Expand $(x-3)(x-2)$.

\textbf{Worked solution:} Multiply each term in the first bracket by each term in the second bracket:
\[
x\cdot x+x\cdot(-2)+(-3)\cdot x+(-3)\cdot(-2)
=x^2-2x-3x+6.
\]
Combine the $x$-terms:
\[
x^2-5x+6.
\]

\textbf{Answer:} $x^2-5x+6$.
\end{frame}

\begin{frame}{Starter 4 Q8}

\textbf{Worked solution:} Use the index law $a^m\times a^n=a^{m+n}$:
\[
a^2\times a=a^{2+1}=a^3.
\]
Then multiply by $b$:
\[
a^3b.
\]

\textbf{Answer:} $a^3b$.
\end{frame}

\begin{frame}[label=starter6sol]{Starter 6 Q1}
\textbf{Question:} Calculate $-4\times 6$.

\textbf{Worked solution:} The signs are different, so the product is negative:
\[
-4\times 6=-(4\times 6)=-24.
\]

\textbf{Answer:} $-24$.

\textbf{Worked solution:} Combine the $x$-terms:
\[
7x+3x=10x.
\]
The constant $-5$ remains unchanged.

\textbf{Answer:} $10x-5$.

\begin{frame}{Starter 6 Q4}

3x=12.

x=4.
\]

\textbf{Answer:} $x=4$.

\begin{frame}{Starter 6 Q5}
\textbf{Question:} Solve $4x+3=2x+15$.

\textbf{Worked solution:} Subtract $2x$ from both sides:
\[
2x+3=15.
\]
Subtract $3$:
\[
2x=12.
\]
Divide by $2$:
\[
x=6.
\]

Check: $4(6)+3=27$ and $2(6)+15=27$.

\textbf{Answer:} $x=6$.
\end{frame}

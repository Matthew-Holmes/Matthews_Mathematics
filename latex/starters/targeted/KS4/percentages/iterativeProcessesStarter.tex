\vspace{0.4cm}

\item A population starts at 200 and grows by $5\%$ each year.

Complete the table.

\[
\begin{array}{c|c}
n & P_n \\
\hline
0 & 200 \\
1 & \ablank{210} \\
2 & \ablank{220.5} \\
3 & \ablank{231.525}
\end{array}
\]

\vspace{0.2cm}


\vspace{0.4cm}

\item An iterative process is
\[
U_{n+1} = U_n + 4, \qquad U_1 = 3
\]

Complete the table.

\[
\begin{array}{c|c}
n & U_n \\
\hline
1 & 3 \\
2 & \ablank{7} \\
3 & \ablank{11} \\
4 & \ablank{15}
\end{array}
\]

\end{enumerate}

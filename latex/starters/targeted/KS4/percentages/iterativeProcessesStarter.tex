\item Find the next two terms:

\]

\vspace{0.4cm}

\item Find an $n$th term for:
\[
8,\ 13,\ 18,\ 23,\ \dots
\]

\vspace{0.4cm}

\item A population starts at 200 and grows by $5\%$ each year.

Complete the table.

\[
\begin{array}{c|c}
n & P_n \\
\hline
0 & 200 \\
1 & \\
2 & \\
3 &
\end{array}
\]

\vspace{0.2cm}


\vspace{0.4cm}

\item An iterative process is
\[
U_{n+1} = U_n + 4, \qquad U_1 = 3
\]

Complete the table.

\[
\begin{array}{c|c}
n & U_n \\
\hline
1 & 3 \\
2 & \\
3 & \\
4 &
\end{array}
\]

\end{enumerate}

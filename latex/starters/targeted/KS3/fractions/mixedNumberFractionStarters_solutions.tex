  \hfill
  \item $\dfrac{3}{4} - \dfrac{1}{2} = \dfrac{3}{4} - \dfrac{2}{4} = \dfrac{1}{4}$
  \hfill
  \item $\dfrac{36}{48} = \dfrac{3}{4}$

  \item $\color{blue}1\dfrac{3}{4} \color{orange}+ 1\dfrac{1}{2} \color{black}= \dfrac{7}{4} + \dfrac{6}{4} = \dfrac{13}{4} = 3\dfrac{1}{4}$

  \item $2\dfrac{1}{3} + 1\dfrac{1}{3} = 3\dfrac{2}{3}$
  \hfill
  $3\dfrac{1}{4} + 1\dfrac{1}{2} = 3\dfrac{1}{4} + 1\dfrac{2}{4} = 4\dfrac{3}{4}$

  \item $1\dfrac{3}{4}$ is at the $1\tfrac{3}{4}$ mark; $3\dfrac{1}{4}$ is at the $3\tfrac{1}{4}$ mark.
  Difference $= 3\dfrac{1}{4} - 1\dfrac{3}{4} = \dfrac{13}{4} - \dfrac{7}{4} = \dfrac{6}{4} = 1\dfrac{1}{2}$

  \item $2\dfrac{2}{3} + 1\dfrac{3}{4} = \dfrac{8}{3} + \dfrac{7}{4} = \dfrac{32}{12} + \dfrac{21}{12} = \dfrac{53}{12} = 4\dfrac{5}{12}$

  \item $1\dfrac{3}{4} - 3\dfrac{1}{4} = \dfrac{7}{4} - \dfrac{13}{4} = -\dfrac{6}{4} = -1\dfrac{1}{2}$
\end{enumerate}
\end{frame}

  ($\dfrac{1}{3} = \dfrac{5}{15}$,\ $\dfrac{2}{5} = \dfrac{6}{15}$,\ so $2\dfrac{2}{5} > 2\dfrac{1}{3}$)

  \item $\dfrac{23}{6} = 3\dfrac{5}{6}$

  \item $3\dfrac{1}{4} - 1\dfrac{3}{4} = \dfrac{13}{4} - \dfrac{7}{4} = \dfrac{6}{4} = 1\dfrac{1}{2}$

  \item $3\dfrac{3}{5} - 1\dfrac{1}{5} = 2\dfrac{2}{5}$
  \hfill
  $4\dfrac{5}{6} - 2\dfrac{1}{6} = 2\dfrac{4}{6} = 2\dfrac{2}{3}$

  \item $4\dfrac{1}{4} - 1\dfrac{1}{2} = \dfrac{17}{4} - \dfrac{6}{4} = \dfrac{11}{4} = 2\dfrac{3}{4}$
  \hfill
  $1\dfrac{2}{3} + 2\dfrac{1}{4} = \dfrac{5}{3} + \dfrac{9}{4} = \dfrac{20}{12} + \dfrac{27}{12} = \dfrac{47}{12} = 3\dfrac{11}{12}$

  \item $3\dfrac{1}{3} - 1\dfrac{3}{4} = \dfrac{10}{3} - \dfrac{7}{4} = \dfrac{40}{12} - \dfrac{21}{12} = \dfrac{19}{12} = 1\dfrac{7}{12}$
  \hfill
  $5\dfrac{1}{3} - 2\dfrac{3}{4} = \dfrac{16}{3} - \dfrac{11}{4} = \dfrac{64}{12} - \dfrac{33}{12} = \dfrac{31}{12} = 2\dfrac{7}{12}$
\end{enumerate}
\end{frame}

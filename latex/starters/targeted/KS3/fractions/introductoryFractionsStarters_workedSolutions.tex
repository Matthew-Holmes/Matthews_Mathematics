\begin{frame}{Worked Solution: Starter 1, Question 1}
\hypertarget{work-st1}{}

\textbf{Question:} Which shapes (A, B, C, D) are divided into equal parts?

\textbf{Worked Solution:} Equal parts must all be the same size. Shape B is a square divided into 4 equal smaller squares; shape C is a circle divided into 4 equal sectors. Shapes A and D have parts of different sizes. \ashow{B and C are divided into equal parts.}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 2}

\textbf{Question:} In shape B, what fraction is shaded?

\textbf{Worked Solution:} Shape B has 4 equal parts, and 1 of those parts is shaded. Therefore the shaded fraction is shaded parts divided by total parts: \ashow{$\frac{1}{4}$}.
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 3}

\textbf{Question:} Draw a rectangle divided into 5 equal parts. Shade $\frac{3}{5}$.

\textbf{Worked Solution:} Divide the rectangle into 5 equal vertical strips, then shade 3 of them.

\ashow{Three out of five equal parts are shaded: $\frac{3}{5}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 4}

\textbf{Question:} Which is larger: $\frac{2}{5}$ or $\frac{3}{5}$?

\textbf{Worked Solution:} The denominators are both 5, so compare the numerators. Since $3>2$, \ashow{$\frac{3}{5}$ is larger.}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 5}

\textbf{Question:} Which is larger: $\frac{3}{5}$ or $\frac{3}{4}$?

\textbf{Worked Solution:} The numerators are both 3. A smaller denominator means the equal parts are larger, so $\frac{3}{4}$ has larger parts than $\frac{3}{5}$. \ashow{$\frac{3}{4}$ is larger.}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 6}

\textbf{Question:} Order the following fractions from smallest to largest: $\frac{1}{100}, \frac{100}{1}, \frac{1}{10}, \frac{10}{1}, \frac{1}{1}$.

\textbf{Worked Solution:} Convert to decimals: $\frac{1}{100}=0.01$, $\frac{1}{10}=0.1$, $\frac{1}{1}=1$, $\frac{10}{1}=10$, $\frac{100}{1}=100$. \ashow{Smallest to largest: $\frac{1}{100},\ \frac{1}{10},\ \frac{1}{1},\ \frac{10}{1},\ \frac{100}{1}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 1}
\hypertarget{work-st2}{}

\textbf{Question:} Write the fraction shaded for each bar.

\textbf{Worked Solution:} Count the shaded strips and the total number of equal strips in each bar. First bar: 1 shaded strip out of 2. Second bar: 2 shaded strips out of 4. Third bar: 3 shaded strips out of 6. \ashow{$\frac{1}{2},\ \frac{2}{4},\ \frac{3}{6}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 2}

\textbf{Question:} Is the same proportion of each bar shaded?

\textbf{Worked Solution:} In each bar, the shaded part is exactly half of the whole bar, even though the bars are split into different numbers of parts. \ashow{Yes, each bar is $\frac{1}{2}$ shaded.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 3}

\textbf{Question:} Use the bars to fill in the blanks: $\frac{\ashowq{1}}{2} = \frac{\ashowq{2}}{4} = \frac{\ashowq{3}}{6}$.

\textbf{Worked Solution:} Each bar is half shaded, so the missing numerators are the numbers that make each fraction equal to $\frac12$. \ashow{Thus the completed equation is $\frac{1}{2}=\frac{2}{4}=\frac{3}{6}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 4}

\textbf{Question:} Shade $\frac{2}{3}$ of a bar divided into 9 parts.

\textbf{Worked Solution:} A bar divided into 9 parts has 3 parts in each third, because $9 \div 3 = 3$. So $\frac23$ means $2 \times 3 = 6$ parts. \ashow{Shade 6 of the 9 parts, i.e. $\frac{6}{9}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 5}

\textbf{Question:} Find the missing numerator: $\frac{2}{3} = \frac{\ashowq{6}}{9}$.

\textbf{Worked Solution:} The denominator $3$ is multiplied by $3$ to get $9$. So the numerator is also multiplied by $3$: $2 \times 3 = \ashow{6}$. \ashow{The missing numerator is 6.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 6}

\textbf{Question:} Find the missing numerator: $\frac{1}{3} = \frac{\ashowq{2}}{6}$.

\textbf{Worked Solution:} The denominator $3$ is multiplied by $2$ to get $6$. So the numerator is also multiplied by $2$: $1 \times 2 = \ashow{2}$. \ashow{The missing numerator is 2.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 7}

\textbf{Question:} Which two fractions are equivalent: $\frac{1}{2}, \frac{2}{5}, \frac{4}{8}, \frac{3}{6}$?

\textbf{Worked Solution:} Simplify $\frac{4}{8}$ by dividing top and bottom by 4, giving $\frac12$. Simplify $\frac{3}{6}$ by dividing top and bottom by 3, giving $\frac12$. The fraction $\frac25$ does not simplify to $\frac12$. \ashow{$\frac{1}{2}=\frac{4}{8}=\frac{3}{6}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 1}
\hypertarget{work-st3}{}

\textbf{Question:} Which is larger: $\frac{1}{2}$ or $\frac{1}{3}$?

\textbf{Worked Solution:} The numerators are the same, so compare the denominators. Since $2<3$, halves are larger than thirds. \ashow{$\frac{1}{2}$ is larger.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 2}

\textbf{Question:} Compare $\frac{2}{5}$ and $\frac{4}{5}$. Which is bigger?

\textbf{Worked Solution:} Both fractions have denominator $5$, so compare numerators: $4>2$. \ashow{$\frac{4}{5}$ is bigger.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 3}

\textbf{Question:} Convert $\frac{3}{4}$ and $\frac{5}{8}$ to a common denominator. Which is larger?

\textbf{Worked Solution:} The denominator $8$ is a multiple of $4$. So $\frac{3}{4} = \frac{3 \times 2}{4 \times 2} = \frac{6}{8}$. Now compare $\frac{6}{8}$ and $\frac{5}{8}$. \ashow{$\frac{3}{4}=\frac{6}{8}$ is larger.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 4}

\textbf{Question:} Order from smallest to largest: $\frac{1}{3}, \frac{1}{6}, \frac{1}{2}$.

\textbf{Worked Solution:} Use denominator $6$: $\frac{1}{3} = \frac{2}{6}$, $\frac{1}{6}$ stays as $\frac{1}{6}$, and $\frac{1}{2} = \frac{3}{6}$. Since $1<2<3$, order by the numerators. \ashow{$\frac{1}{6},\ \frac{1}{3},\ \frac{1}{2}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 5}

\textbf{Question:} Sam ate $\frac{2}{3}$ of a pizza, Lee ate $\frac{5}{6}$ of the same pizza. Who ate more?

\textbf{Worked Solution:} Use denominator $6$: $\frac{2}{3} = \frac{4}{6}$. Since $\frac{5}{6} > \frac{4}{6}$, Lee ate more. \ashow{Lee ate more.}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 6}

\textbf{Question:} Compare $\frac{4}{6}$ and $\frac{2}{3}$.

\textbf{Worked Solution:} Simplify $\frac{4}{6}$ by dividing numerator and denominator by $2$: $\frac{4 \div 2}{6 \div 2} = \frac{2}{3}$. \ashow{Equal: $\frac{4}{6}=\frac{2}{3}$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 1}
\hypertarget{work-st4}{}

\textbf{Question:} A cake is cut into 10 equal slices. If 3 are eaten, what fraction remains?

\textbf{Worked Solution:} The whole cake has 10 slices. If 3 are eaten, the number remaining is $10-3=7$. So the fraction remaining is \ablank{$\frac{7}{10}$}.
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 2}

\textbf{Question:} There are 30 students. $\frac{2}{5}$ walk to school. How many walk?

\textbf{Worked Solution:} First find one fifth: $30 \div 5 = 6$. Two fifths is two of these parts: $2 \times 6 = 12$. \ashow{12 students walk.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 3}

\textbf{Question:} A ruler is 24 cm long. Mark $\frac{1}{2}$ and $\frac{1}{3}$ on it. How many cm is each?

\textbf{Worked Solution:} $\frac13$ of 24 cm means $24 \div 3 = 8$ cm. $\frac12$ of 24 cm means $24 \div 2 = 12$ cm. \ashow{$\frac13 = 8$ cm and $\frac12 = 12$ cm.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 4}

\textbf{Question:} Which is longer: $\frac{3}{4}$ of an hour or $\frac{2}{3}$ of an hour? (1 hour = 60 mins)

\textbf{Worked Solution:} $\frac34$ of 60 mins: $60 \div 4 \times 3 = 45$ mins. $\frac23$ of 60 mins: $60 \div 3 \times 2 = 40$ mins. Since $45 > 40$, \ashow{$\frac34$ hour is longer.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 5}

\textbf{Question:} You have \$30. You spend $\frac{2}{5}$ on a book. How much do you spend? How much left?

\textbf{Worked Solution:} One fifth of \$30 is $30 \div 5 = \$6$. Two fifths is $2 \times \$6 = \$12$. The amount left is $30 - 12 = \$18$. \ashow{Spend \$12, left \$18.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 6}

\textbf{Question:} A recipe needs $\frac{3}{4}$ cup of milk. You only have a $\frac{1}{4}$ cup measure. How many times must you fill it?

\textbf{Worked Solution:} Count quarter-cups: $\frac14 + \frac14 + \frac14 = \frac34$. \ashow{You must fill the $\frac14$ cup 3 times.}
\end{frame}

\begin{frame}{Worked Solution: Starter 9, Question 5}

\textbf{Question:} Convert to mixed numbers: $\frac{17}{6},\ \frac{22}{9}$.

\textbf{Worked Solution:} $17 \div 6 = 2$ remainder $5$, so $\frac{17}{6}=2\frac56$. $22 \div 9 = 2$ remainder $4$, so $\frac{22}{9}=2\frac49$. \ashow{$2\frac56,\ 2\frac49$.}
\end{frame}

\begin{frame}{Worked Solution: Starter 9, Question 6}

\textbf{Question:} Convert to improper fractions: $4\frac{1}{3},\ 2\frac{5}{6}$.

\textbf{Worked Solution:} $4\frac13 = \frac{3\times4+1}{3} = \frac{13}{3}$. $2\frac56 = \frac{6\times2+5}{6} = \frac{17}{6}$. \ashow{$\frac{13}{3},\ \frac{17}{6}$.}
\end{frame}

\textbf{Worked Solution:} Both mixed numbers have whole part 2, so compare the fractional parts $\frac14$ and $\frac25$. Use denominator 20: $\frac14=\frac{5}{20}$ and $\frac25=\frac{8}{20}$. Since $\frac{8}{20}>\frac{5}{20}$, \ashow{$2\frac25$ is larger.}

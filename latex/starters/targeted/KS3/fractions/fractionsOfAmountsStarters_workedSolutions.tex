\begin{frame}{Worked Solution: Starter 1, Question 1}
\hypertarget{work-st1}{}
\textbf{Question:} Fully simplify \(\dfrac{18}{24}\).

\awork{Solution: The highest common factor of \(18\) and \(24\) is \(6\), so divide numerator and denominator by \(6\):
\[
\frac{18}{24}=\frac{18\div 6}{24\div 6}=\frac{3}{4}.
\]
So \(\dfrac{18}{24}\) simplifies to \(\dfrac{3}{4}\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 2}
\textbf{Question:} Calculate \(2\dfrac{1}{3}+1\dfrac{1}{2}\).

\awork{Solution: Use a common denominator of \(6\):
\[
2\frac13=2\frac26,\qquad 1\frac12=1\frac36.
\]
\[
2\frac26+1\frac36=3\frac56.
\]
So the sum is \(3\dfrac{5}{6}\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 3}
\textbf{Question:} Using the bar models, find \(\dfrac12\) of \(20\), \(\dfrac14\) of \(20\), and \(\dfrac34\) of \(20\).

\awork{Solution: The bar models split the whole \(20\) into equal parts.

\(\dfrac12\) of \(20\): split into \(2\) parts of \(10\), so \(10\).

\(\dfrac14\) of \(20\): split into \(4\) parts of \(5\), so \(5\).

\(\dfrac34\) of \(20\): take \(3\) of the \(4\) equal parts, so \(3\times 5=15\).

Therefore:
\[
\frac12\text{ of }20=10,\qquad
\frac14\text{ of }20=5,\qquad
\frac34\text{ of }20=15.
\]}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 4}
\textbf{Question:} Calculate \(\dfrac13\) of \(24\) and \(\dfrac23\) of \(24\).

\awork{Solution: First find one third:
\[
24\div 3=8.
\]
So \(\dfrac13\) of \(24=8\).

Two thirds are two lots of one third:
\[
\dfrac23\text{ of }24=2\times 8=16.
\]
Hence the answers are \(8\) and \(16\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 5}
\textbf{Question:} Fill in the blank: \(\dfrac34\) of \(\_\_\_ = 18\).

\awork{Solution: Let the missing number be \(x\). We need
\[
\frac34 x=18.
\]
Divide both sides by \(3\): one quarter of \(x\) is \(6\). Then multiply by \(4\): the whole of \(x\) is \(4\times 6=24\).

So the blank is \(24\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 6}
\textbf{Question:} Find \(\dfrac25\) of \(35\).

\awork{Solution: Divide \(35\) into \(5\) equal parts:
\[
35\div 5=7.
\]
Each fifth is \(7\), so two fifths are
\[
2\times 7=14.
\]
Therefore \(\dfrac25\) of \(35=14\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 7}
\textbf{Question:} Amy cuts a \(60\,\text{cm}\) ribbon and takes \(\dfrac34\) of it. How long is her piece?

\awork{Solution: Divide the ribbon into \(4\) equal parts:
\[
60\div 4=15.
\]
Each quarter is \(15\,\text{cm}\). Amy takes \(3\) quarters, so her piece is
\[
3\times 15=45\,\text{cm}.
\]
So Amy's piece is \(45\,\text{cm}\) long.}
\end{frame}

\begin{frame}{Worked Solution: Starter 1, Question 8}
\textbf{Question:} True or false? ``\(\dfrac12\) of \(30=30\div 2\).'' Explain your reasoning.

\awork{Solution: True. The word ``of'' means multiply, so
\[
\frac12\text{ of }30=\frac12\times 30=\frac{30}{2}=15.
\]
Also,
\[
30\div 2=15.
\]
Since both give \(15\), the statement is true.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 1}
\hypertarget{work-st2}{}
\textbf{Question:} Order smallest to largest: \(\dfrac38,\ \dfrac12,\ \dfrac{5}{12}\).

\awork{Solution: Use a common denominator of \(24\):
\[
\frac38=\frac{9}{24},\qquad
\frac12=\frac{12}{24},\qquad
\frac{5}{12}=\frac{10}{24}.
\]
The numerators are \(9<10<12\), so
\[
\frac38<\frac{5}{12}<\frac12.
\]}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 2}
\textbf{Question:} Convert \(\dfrac{17}{5}\) to a mixed number.

\awork{Solution: Divide \(17\) by \(5\). Since \(5\times 3=15\), there are \(3\) whole fives with remainder \(2\):
\[
\frac{17}{5}=3\frac{2}{5}.
\]
So the mixed number is \(3\dfrac25\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 3}
\textbf{Question:} Use the bar model to find \(\dfrac34\) of \(48\) and \(\dfrac58\) of \(48\).

\awork{Solution: For \(\dfrac34\) of \(48\): split \(48\) into \(4\) equal parts.
\[
48\div 4=12,\qquad 3\times 12=36.
\]
For \(\dfrac58\) of \(48\): split \(48\) into \(8\) equal parts.
\[
48\div 8=6,\qquad 5\times 6=30.
\]
So the answers are \(36\) and \(30\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 4}
\textbf{Question:} Calculate \(\dfrac58\) of \(72\).

\awork{Solution: Divide \(72\) into \(8\) equal parts:
\[
72\div 8=9.
\]
Each eighth is \(9\), so five eighths are
\[
5\times 9=45.
\]
Therefore \(\dfrac58\) of \(72=45\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 5}
\textbf{Question:} \(\dfrac23\) of a class of \(30\) are girls. How many boys are there?

\awork{Solution: If \(\dfrac23\) are girls, then \(\dfrac13\) are boys.
\[
30\div 3=10.
\]
So there are \(10\) boys.

Alternatively, girls \(=\dfrac23\times 30=20\), so boys \(=30-20=10\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 6}
\textbf{Question:} Rania says ``\(\dfrac35\) of \(40=24\)''. Is she correct? Show your working.

\awork{Solution: Split \(40\) into \(5\) equal parts:
\[
40\div 5=8.
\]
Three fifths are
\[
3\times 8=24.
\]
So Rania is correct: \(\dfrac35\) of \(40=24\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 7}
\textbf{Question:} Which is greater: \(\dfrac34\) of \(60\), or \(\dfrac45\) of \(55\)?

\awork{Solution: Compute each amount.
\[
\frac34\times 60=45.
\]
\[
\frac45\times 55=\frac{4\times 55}{5}=\frac{220}{5}=44.
\]
Since \(45>44\), \(\dfrac34\) of \(60\) is greater.}
\end{frame}

\begin{frame}{Worked Solution: Starter 2, Question 8}
\textbf{Question:} Fill in the blank:
\[
\frac{\_\_\_}{8}\text{ of }64=40.
\]

\awork{Solution: Let the missing numerator be \(x\). Then
\[
\frac{x}{8}\times 64=40.
\]
Since \(64\div 8=8\),
\[
8x=40.
\]
\[
x=40\div 8=5.
\]
So the missing number is \(5\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 2}
\textbf{Question:} Use the area model to calculate
\[
\frac12\times\frac13=\frac{\_\_\_}{\_\_\_}.
\]

\awork{Solution: The area model is divided into \(2\) equal columns and \(3\) equal rows, so it has
\[
2\times 3=6
\]
equal parts. The overlap of \(\dfrac12\) and \(\dfrac13\) is exactly \(1\) of those \(6\) parts.

Therefore
\[
\frac12\times\frac13=\frac16.
\]
The missing numbers are \(1\) and \(6\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 3}
\textbf{Question:} Use the second area model to calculate and simplify:
\[
\frac23\times\frac34=\frac{\_\_\_}{\_\_\_}=\frac{\_\_\_}{\_\_\_}.
\]

\awork{Solution: The second area model is divided into \(3\) equal columns and \(4\) equal rows, giving
\[
3\times 4=12
\]
equal parts. The overlap is \(2\times 3=6\) parts, so
\[
\frac23\times\frac34=\frac{6}{12}.
\]
Now simplify by dividing numerator and denominator by \(6\):
\[
\frac{6}{12}=\frac12.
\]
So the completed fraction is
\[
\frac23\times\frac34=\frac{6}{12}=\frac12.
\]}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 4}

\frac35\times\frac57.

\awork{Solution: Multiply numerators and denominators:
\[
\frac35\times\frac57=\frac{3\times 5}{5\times 7}=\frac{15}{35}.
\]
Simplify by dividing numerator and denominator by \(5\):
\[
\frac{15}{35}=\frac37.
\]
So \(\dfrac35\times\dfrac57=\dfrac37\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 3, Question 5}

\awork{Solution: First simplify each fraction:

Then multiply:

\frac13\times\frac12=\frac{1\times 1}{3\times 2}=\frac16.
\]
So \(\dfrac39\times\dfrac48=\dfrac16\).}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 2}

\begin{frame}{Worked Solution: Starter 4, Question 6}
\textbf{Question:} Which is greater: \(\dfrac23\times\dfrac34\), or \(\dfrac34\) of \(\dfrac23\)? Explain.

\awork{Solution: Since ``of'' means multiply, both are products of the same two fractions:
\[
\frac23\times\frac34=\frac{2\times 3}{3\times 4}=\frac{6}{12}=\frac12,
\]
and
\[
\frac34\text{ of }\frac23=\frac34\times\frac23=\frac{3\times 2}{4\times 3}=\frac{6}{12}=\frac12.
\]
They are equal. Multiplication of fractions is commutative, so neither is greater.}
\end{frame}

\begin{frame}{Worked Solution: Starter 4, Question 7}
\textbf{Question:} Fill in the blank:
\[
\frac35\times \_\_\_ = \frac{6}{35}.
\]

\awork{Solution: Let the missing fraction be \(x\). To find \(x\), divide \(\dfrac{6}{35}\) by \(\dfrac35\):
\[
x=\frac{6}{35}\div\frac35=\frac{6}{35}\times\frac53
=\frac{6\times 5}{35\times 3}
=\frac{30}{105}
=\frac{2}{7}.
\]
So the missing fraction is \(\dfrac27\).}

\begin{frame}{Worked Solution: Starter 4, Question 8}
\textbf{Question:} True or false? ``Multiplying two proper fractions always gives a smaller result than either fraction.'' Explain or give a counterexample.

\awork{Solution: True for positive proper fractions. A proper fraction lies between \(0\) and \(1\), and multiplying a positive number by a number less than \(1\) makes it smaller.

For example,
\[
\frac23\times\frac34=\frac{6}{12}=\frac12.
\]
Here \(\dfrac12<\dfrac23\) and \(\dfrac12<\dfrac34\), so the result is smaller than both fractions. Therefore the statement is true.}
\end{frame}

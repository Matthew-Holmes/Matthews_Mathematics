  \begin{block}{4) What is the area of the rectangle?}
    \begin{tikzpicture}[scale=0.6]

% Outer rectangle
\draw (0,0) rectangle (6,2);

% Vertical split
\draw (3,0) -- (3,2);

% Labels for widths
\node at (1.5,-0.3) {$f$};
\node at (4.5,-0.3) {$6$};

% Height label
\node at (-0.3,1) {$x$};

% Area labels
\node at (1.5,1) {\ashowq{$xf$}};
\node at (4.5,1) {\ashowq{$6x$}};

\end{tikzpicture}
    \par\vspace{0.5em}
    \ashow{Area = \(x(f+6)\)}
  \end{block}
\end{minipage}

% ---------- ANSWERS ----------
\begin{frame}{Answers}
\begin{enumerate}
  \item \(3(y+5)=3y+15\)
  \item \(a(b-5)=ab-5a\)
  \item \((x+4)(x-2)=x^2+2x-8\)
  \item Area = \(xf + 6x = x(f+6)\)
\end{enumerate}
\end{frame}

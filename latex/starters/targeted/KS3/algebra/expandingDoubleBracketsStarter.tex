% ---------- ANSWERS ----------
\begin{frame}{Answers}
\begin{enumerate}
  \item \(3(y+5)=3y+15\)
  \item \(a(b-5)=ab-5a\)
  \item \((x+4)(x-2)=x^2+2x-8\)
  \item Area = \(xf + 6x = x(f+6)\)
\end{enumerate}

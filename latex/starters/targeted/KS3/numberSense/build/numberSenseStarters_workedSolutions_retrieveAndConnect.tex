% Worked solutions for Starter 1
\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 1}
\hypertarget{work-st1}{}
\textbf{Question:} On Numberline A, what numbers are marked by \(P\) and \(Q\)?
\awork{Solution: Numberline A goes from \(0\) to \(10\) in ten equal steps. \(P\) is three steps from \(0\), so \(P=3\). \(Q\) is eight steps from \(0\), so \(Q=8\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 2}
\textbf{Question:} On Numberline B, what numbers are marked by \(S\) and \(T\)?
\awork{Solution: Numberline B is divided into ten equal steps from \(0\) to \(1\), so each step is \(0.1\). \(S\) is three steps from \(0\), so \(S=0.3\). \(T\) is nine steps from \(0\), so \(T=0.9\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 3}
\textbf{Question:} Mark \(6\) on Line A and \(0.45\) on Line B.
\awork{Solution: On Line A, each gap is \(1\), so \(6\) is one gap to the right of the tick labelled \(5\). On Line B, \(0.45\) is halfway between \(0.4\) and \(0.5\), so place the mark halfway between those two ticks.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 4}
\textbf{Question:} Sam says \(P=4\), as it is the \(4\)th mark. What has he done wrong?
\awork{Solution: Sam is counting the tick at \(0\) as the first mark, so he calls the tick three steps later the fourth mark. But a coordinate is measured by the number of steps from \(0\), not by counting marks. \(P\) is three whole steps from \(0\), so \(P=3\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 5}
\textbf{Question:} The chart shows \(407.52\). What is the \(4\) worth? And the \(5\)?
\awork{Solution: In \(407.52\), the \(4\) is in the hundreds column, so it is worth \(4\times100=400\). The \(5\) is in the tenths column, so it is worth \(5\times0.1=0.5\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 6}
\textbf{Question:} Work out \(407.52\times10\) and \(407.52\div10\).
\awork{Solution: Multiplying by \(10\) moves every digit one place to the left, so \(407.52\times10=4075.2\). Dividing by \(10\) moves every digit one place to the right, so \(407.52\div10=40.752\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 7}
\textbf{Question:} A number line goes \(3.6\) to \(3.7\) in \(10\) equal steps. What is \(3\) steps along from \(3.6\)?
\awork{Solution: The interval from \(3.6\) to \(3.7\) has length \(0.1\). Ten equal steps make each step \(0.1\div10=0.01\). Three steps is \(0.03\), so the number is \(3.6+0.03=3.63\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 1, Question 8}
\textbf{Question:} Use \(7\), \(5\), \(4\), \(2\) and a decimal point to make the largest number less than \(100\).
\awork{Solution: To stay below \(100\), use a two-digit whole-number part. Put the largest digit, \(7\), in the tens column, and the next largest, \(5\), in the ones column, giving \(75\). Then arrange the remaining digits as the larger fraction, \(.42\), rather than \(.24\). The largest number is \(75.42\).}
\end{frame}

% Worked solutions for Starter 2
\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 1}
\hypertarget{work-st2}{}
\textbf{Question:} Fill in \(<\) or \(>\): \(7 \ \_ \ 12\) and \(30 \ \_ \ 3\).
\awork{Solution: On a number line, the smaller number is to the left. Since \(7\) is to the left of \(12\), \(7<12\). Since \(3\) is to the left of \(30\), \(30>3\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 2}
\textbf{Question:} In \(0.405\), what is the \(4\) worth?
\awork{Solution: In a decimal, the first place after the decimal point is the tenths place. The \(4\) is in the tenths place, so it is worth \(4\) tenths, or \(0.4\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 3}
\textbf{Question:} On the place value chart, which is bigger, \(0.4\) or \(0.35\)? Why?
\awork{Solution: Both numbers have \(0\) ones. Compare the tenths: \(0.4\) has \(4\) tenths, while \(0.35\) has only \(3\) tenths. Therefore \(0.4>0.35\); the extra hundredths in \(0.35\) cannot make up for having fewer tenths.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 4}
\textbf{Question:} Fill in \(<\), \(>\), or \(=\): \(0.4 \ \_ \ 0.40\) and \(0.6 \ \_ \ 0.06\).
\awork{Solution: Writing an extra zero at the end of a decimal does not change its value, so \(0.4=0.40\). Since \(0.6\) has \(6\) tenths but \(0.06\) has \(0\) tenths, \(0.6>0.06\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 5}
\textbf{Question:} Mark \(0.38\) on Line A. Is it nearer \(0.3\) or \(0.4\)?
\awork{Solution: \(0.38\) is \(0.08\) above \(0.3\), but only \(0.02\) below \(0.4\). So mark it two hundredths to the left of \(0.4\); it is nearer \(0.4\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 6}
\textbf{Question:} Order, smallest first: \(0.5\), \(0.45\), \(0.405\), \(0.54\).
\awork{Solution: Compare using the same place values: \(0.405=0.405\), \(0.45=0.450\), \(0.5=0.500\), and \(0.54=0.540\). Hence the order is \(0.405<0.45<0.5<0.54\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 7}
\textbf{Question:} True or false: more digits \emph{after} the decimal point means a bigger number? Give a counter-example.
\awork{Solution: False. More decimal digits do not automatically make a number larger, because place value matters first. A counter-example is \(0.9>0.1234\): \(0.9\) has only one decimal digit, but it has \(9\) tenths, while \(0.1234\) has only \(1\) tenth.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 2, Question 8}
\textbf{Question:} Alfie says \(0.45>0.5\) as \(45>5\). Explain his mistake, then write a decimal between \(0.4\) and \(0.405\).
\awork{Solution: Alfie is comparing \(45\) and \(5\) as whole numbers and ignoring place value. In hundredths, \(0.45=45\) hundredths and \(0.5=50\) hundredths, so \(0.5>0.45\). One decimal between \(0.4=0.400\) and \(0.405\) is \(0.402\), since \(0.400<0.402<0.405\).}
\end{frame}

% Worked solutions for Starter 3
\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 1}
\hypertarget{work-st3}{}
\textbf{Question:} Which is colder, \(4\,^\circ\)C or \(-4\,^\circ\)C?
\awork{Solution: \(4\,^\circ\)C is above freezing, while \(-4\,^\circ\)C is below freezing. The colder temperature is the lower one, so \(-4\,^\circ\)C is colder.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 2}
\textbf{Question:} What is each mark on the thermometer worth? Write the temperatures at A and B.
\awork{Solution: The thermometer goes from \(-10\) to \(10\) with ten equal gaps, so each mark is worth \(2\,^\circ\)C. A is three gaps below \(0\), so \(A=-6\,^\circ\)C. B is two gaps above \(0\), so \(B=4\,^\circ\)C.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 3}
\textbf{Question:} Which is colder, \(-3\) or \(-8\)? Fill in: \(-3 \ \_ \ -8\).
\awork{Solution: On a number line, \(-8\) is to the left of \(-3\), so \(-8\) is smaller and colder. Therefore \(-3>-8\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 4}
\textbf{Question:} Order, smallest first: \(-2\), \(5\), \(-7\), \(0\), \(-1\).
\awork{Solution: Negative numbers are smaller than \(0\), and \(5\) is the only positive number. Compare \(-7,-2,-1\): the most negative is \(-7\), then \(-2\), then \(-1\). The order is \(-7<-2<-1<0<5\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 5}
\textbf{Question:} Mark \(-0.5\) and \(-0.05\) on Line C. Which is larger?
\awork{Solution: On a number line, the larger number is to the right. \(-0.5\) is fifty hundredths below zero, while \(-0.05\) is only five hundredths below zero. Mark \(-0.5\) halfway between \(-1\) and \(0\), and mark \(-0.05\) much closer to \(0\). Hence \(-0.05\) is larger.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 6}
\textbf{Question:} Fill in \(<\) or \(>\): \(-4.5 \ \_ \ -4.05\) and \(-0.3 \ \_ \ 0.03\).
\awork{Solution: Write \(-4.5=-4.50\). Since \(-4.50\) is farther below \(0\) than \(-4.05\), \(-4.5<-4.05\). Also, every negative number is smaller than every positive number, so \(-0.3<0.03\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 3, Question 7}

\textbf{Question:} Round to the nearest whole number: \(4.6\), \(12.348\).
\awork{Solution: Look at the tenths digit. For \(4.6\), the tenths digit is \(6\geq5\), so \(4.6\to5\). For \(12.348\), the tenths digit is \(3<5\), so \(12.348\to12\).}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 3}
\textbf{Question:} Mark \(2.763\) on Line A. Round it to \(1\) decimal place.
\awork{Solution: For \(1\) d.p., compare \(2.7\) and \(2.8\). Since \(2.763-2.7=0.063\) and \(2.8-2.763=0.037\), it is nearer \(2.8\). Mark it between \(2.76\) and \(2.77\) on the number line; \(2.763\to2.8\).}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 4}
\textbf{Question:} Round to \(1\) d.p.: \(4.62\), \(0.95\). Why is \(0.95\) not \(1\)?
\awork{Solution: For \(4.62\), look at the hundredths digit \(2<5\), so \(4.62\to4.6\). \(0.95\) is halfway between \(0.9\) and \(1.0\), so with the usual round-half-up convention \(0.95\to1.0\). We write \(1.0\), not \(1\), because a number rounded to \(1\) d.p. must show one digit after the decimal point.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 5}
\textbf{Question:} Round \(12.348\) to \(2\) d.p., then to \(1\) d.p. Why is it not \(12.4\)?
\awork{Solution: To \(2\) d.p., look at the third decimal digit, \(8\geq5\), so \(12.348\to12.35\). To \(1\) d.p., look at the second decimal digit of the original number, \(4<5\), so \(12.348\to12.3\). It is not \(12.4\), because that would come from rounding \(12.35\) again; never round in stages.}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 6}
\textbf{Question:} Line B shows \(\frac23=0.\dot6\). Round it to \(1\) d.p.\ and to \(2\) d.p.
\awork{Solution: \(\frac23=0.6666\ldots\). To \(1\) d.p., the second decimal digit is \(6\geq5\), so it rounds up to \(0.7\). To \(2\) d.p., the third decimal digit is \(6\geq5\), so it rounds up to \(0.67\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 7}
\textbf{Question:} Write \(\frac13\) in dot notation, then round \(\frac17\) to \(3\) d.p.
\awork{Solution: \(\frac13=0.333\ldots\), so in dot notation it is \(0.\dot{3}\). Also \(\frac17=0.142857\ldots\). To \(3\) d.p., look at the fourth decimal place: the digits are \(1428\), and \(8\geq5\), so \(0.142857\ldots\to0.143\).}
\end{frame}

\begin{frame}{Worked Solution: Retrieve and Connect 5, Question 8}
\textbf{Question:} Which listed decimals never settle into a repeating block? Can one still be a fraction?
\awork{Solution: \(\pi\), \(\sqrt2\), and \(0.1010010001\ldots\) never settle into a repeating block, so they cannot be written as fractions. Repeating decimals, however, can still be fractions: for example, \(0.\dot3=\frac13\), and from the table \(0.\dot6=\frac23\).}

\section*{Factors}
\textbf{True or False:}
\begin{enumerate}[label=\arabic*.]
  \item \textbf{True.} \(21=7\times 3\).
  \item \textbf{True.} Every number \(n\) can be written as \(1\times n\).
  \item \textbf{False.} Factors of \(16\) are \(1,2,4,8,16\).
  \item \textbf{False.} Square numbers have an odd number of factors; for example \(9\) has \(1,3,9\).
  \item \textbf{False.} \(4\) is a factor of \(12\), but \(8\) is not.
\end{enumerate}

\textbf{Quick questions:}
\begin{enumerate}[label=\arabic*.]
  \item \(1,2,3,4,6,12\)
  \item \(1,2,4,5,10,20\)
  \item \(1,2,3,4,6,9,12,18,36\)
  \item Five factors: \(1,2,4,8,16\)
  \item \(\mathrm{HCF}=6\)
  \item \(\mathrm{HCF}=12\)
  \item \(1\times 48,\ 2\times 24,\ 3\times 16,\ 4\times 12,\ 6\times 8\)
  \item \(24\), with eight factors.
  \item Any prime squared: e.g.\ \(4,\ 9,\ 25\).
  \item Factors pair up, but the square root pairs with itself, leaving one unpaired factor.
\end{enumerate}

  \item \textbf{True.}
  \item \textbf{False.} \(51=3\times 17\).
  \item \textbf{True.}
  \item \textbf{False.} \(33=3\times 11\).
\end{enumerate}

\textbf{Quick questions:}
\begin{enumerate}[label=\arabic*.]
  \item \(2,3,5,7\)
  \item \(11,13,17,19\)
  \item No: \(27=3\times 9\).
  \item \(31\)
  \item No: \(91=7\times 13\).
  \item Four: \(23,29,31,37\)

\section*{Square numbers from prime factors}
\textbf{True or False:}
\begin{enumerate}[label=\arabic*.]
  \item \textbf{True.}
  \item \textbf{False.} The power of \(2\) is odd.
  \item \textbf{True.} \(2^4\times 5^6=(2^2\times 5^3)^2\).
  \item \textbf{False.} Square numbers have an odd number of factors.
  \item \textbf{True.} Multiplying by \(4=2^2\) keeps all powers even.
\end{enumerate}

\textbf{Quick questions:}
\begin{enumerate}[label=\arabic*.]
  \item \(36=2^2\times 3^2=(2\times 3)^2\).
  \item Yes; square root \(=30\).
  \item No, because the power of \(2\) is odd.
  \item \(2^3\times 3^2=72\)
  \item Yes: \(400=2^4\times 5^2=20^2\).
  \item No: \(150=2\times 3\times 5^2\), with odd powers of \(2\) and \(3\).
  \item \(2^4\times 3\times 5^2=1200\)
  \item \(k=6\)
  \item Multiply by \(14\); \(504\times 14=84^2\).
  \item Divide by \(30\); \(1080\div 30=6^2\).
\end{enumerate}

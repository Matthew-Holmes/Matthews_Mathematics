\begin{enumerate}[label=\textbf{Q\arabic*.}]

\item The numerator is above the line and the denominator below it.
\begin{enumerate}[label=(\alph*)]
    \item numerator 3, denominator 8
    \item numerator 5, denominator 12
\end{enumerate}

\item Divide the numerator and the denominator by the same factor.
\begin{enumerate}[label=(\alph*)]
    \item \( \frac{6}{9} = \frac{6 \div 3}{9 \div 3} = \frac{2}{3} \)
    \item \( \frac{20}{25} = \frac{20 \div 5}{25 \div 5} = \frac{4}{5} \)
\end{enumerate}

\item Multiplying the denominator by 4 gives 12, so multiply the numerator by 4 as well:
\( \frac{2}{3} = \frac{2 \times 4}{3 \times 4} = \frac{8}{12} \).

\end{enumerate}

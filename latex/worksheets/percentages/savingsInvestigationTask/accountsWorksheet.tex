{\bfseries\large Part 1}
\begin{enumerate}[leftmargin=8mm,itemsep=2mm,topsep=2mm]
  \item \textbf{Amara, \pounds100, 1 year --- Metroline, \pounds110.}
        Coventree and Starlight both give \pounds105; Chace \pounds104. Virgo needs \pounds500, Tesbury needs \pounds200.
  \item \textbf{Ben, \pounds200, 2 years --- Metroline, \pounds240.}
        Tesbury \pounds224, Starlight \pounds220.50, Coventree \pounds220, Chace \pounds216.32.
  \item \textbf{Chloe, \pounds400, 4 years --- Tesbury, \pounds496.}
        Metroline is out: \pounds400 is over its \pounds300 maximum. Starlight \pounds486.20, Coventree \pounds480, Chace \pounds467.94.
  \item \textbf{Dev, \pounds1000, 3 years --- Virgo, \pounds1259.71.}
        Tesbury \pounds1180, Starlight \pounds1157.63, Coventree \pounds1150, Chace \pounds1124.86.
  \item \textbf{Erin, \pounds500, 5 years --- Virgo, \pounds734.66.}
        Tesbury \pounds650, Starlight \pounds638.14, Coventree \pounds625, Chace \pounds608.33.
  \item \textbf{Fay --- Starlight.} \pounds289.41 against Coventree's \pounds287.50, so \pounds1.91 better off.
\end{enumerate}

\vspace{4mm}
{\bfseries\large Part 2}
\begin{enumerate}[leftmargin=8mm,itemsep=2mm,topsep=2mm]
  \item \textbf{Priya, \pounds200, 1 year --- NatEast, \pounds218.}
        Monza \pounds212.34, Barcourt \pounds210, Halifield \pounds208.08. Santandra needs 2 years; Nationworth needs \pounds250.
  \item \textbf{Marcus, \pounds1000, 2 years --- Santandra, \pounds1166.40.}
        Monza \pounds1127.16, Barcourt \pounds1102.50, Halifield \pounds1082.43, Nationworth \pounds1072. He is too old for NatEast.
  \item \textbf{Sofia, \pounds150, 3 years --- Lloydale, \pounds188.91} (\pounds163.91 plus the \pounds25 bonus).
        Monza \pounds179.50, Barcourt \pounds173.64, Halifield \pounds168.92, Trustbridge \pounds168. Santandra needs \pounds200.
  \item \textbf{Dan, \pounds2000, 2 years --- Santandra, \pounds2332.80.}
        Barcourt \pounds2205, Nationworth \pounds2192, Halifield \pounds2164.86. Monza is capped at \pounds1000.
  \item \textbf{Aisha, \pounds500, 1 year --- Monza, \pounds530.84.}
        Barcourt \pounds525, Halifield \pounds520.20, Nationworth \pounds506. Santandra needs 2 years, Lloydale needs 3.
  \item \textbf{Leo, \pounds300, 3 years --- NatEast, \pounds381,} just ahead of \textbf{Santandra, \pounds377.91} --- a difference of \textbf{\pounds3.09}.
        Then Monza \pounds359, Lloydale \pounds352.82, Barcourt \pounds347.29, Halifield \pounds337.85, Trustbridge \pounds336, Nationworth \pounds282.
\end{enumerate}

\vspace{4mm}
{\bfseries\large Think about}
\begin{itemize}[leftmargin=5mm,itemsep=2mm,topsep=2mm]
  \item \textbf{Coventree v Starlight.} Year 1 is identical: 5\% of the same starting amount. After that Starlight charges its 5\% on a larger balance every year, so it can never fall behind.
  \item \textbf{Biggest percentage.} No. Metroline's 10\% is the highest rate in part 1, but the \pounds300 cap makes it useless to Chloe, Dev and Erin.
  \item \textbf{Trustbridge.} The rates are simple, worked out on the original deposit, so three years pays 2\% + 4\% + 6\% = 12\% in total --- only 4\% a year on average.
  \item \textbf{Nationworth.} The \pounds2 monthly fee costs \pounds24 a year. On \pounds250 that fee wipes out more than the interest; you need over \pounds400 in the account before 6\% even covers it.
  \item \textbf{Halifield.} No. Two lots of 2\% gives 1.02 $\times$ 1.02 = 1.0404, so 4.04\% a year, slightly more than 4\%.
\end{itemize}

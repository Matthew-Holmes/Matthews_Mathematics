% \barmodel{<cells>}{<cells under the lower brace>}{<lower label>}{<upper label>}{<width in cm>}
% The same bar model as the Quick Questions slides: the whole bar

\newcommand{\barmodel}[5]{%
  \begin{tikzpicture}[x=1cm,y=1cm]
    \pgfmathsetmacro{\cw}{#5/#1}
    \foreach \i in {1,...,#1}
      {\draw[gray!70] ({(\i-1)*\cw},0) rectangle ({\i*\cw},0.42);}
    \draw[thick] (0,0) rectangle (#5,0.42);
    \draw[decorate,decoration={brace,amplitude=3pt}] (0,0.54) -- (#5,0.54)
      node[midway,above=2pt,font=\small]{#4};
    \draw[decorate,decoration={brace,mirror,amplitude=3pt}] (0,-0.1) -- ({#2*\cw},-0.1)
      node[midway,below=2pt,font=\small]{#3};
  \end{tikzpicture}%
}

\item
\begin{minipage}[t]{0.63\linewidth}
$50$ pupils voted for the theme of the summer disco.
\begin{enumerate}
  \item What percentage voted for Neon? \ans{$36\%$}
        \work{$\frac{18}{50}=\frac{36}{100}=36\%$}
  \item What percentage voted for Jungle? \ans{$10\%$}
        \work{$\frac{5}{50}=\frac{10}{100}=10\%$}
  \item Ravi says ``more than half voted for Space or Retro''.
        Is Ravi right? Explain why or why not, including a calculation to back up your answer.
        \work{Yes. Space and Retro together are $15+12=27$ votes,
              and $\frac{27}{50}=\frac{54}{100}=54\%$, which is more
              than half.}
\end{enumerate}
\end{minipage}\hfill

%----------------------------------------------------------------
\item Work out the new price of each item.
\begin{multicols}{2}
\begin{enumerate}
  \item Trainers cost $\pounds 45$ and are reduced by $20\%$ in a
        sale. \ans{$\pounds 36$}
        \work{$10\%$ is $\pounds 4.50$, so $20\%$ is $\pounds 9$.\\
              $\pounds 45 - \pounds 9 = \pounds 36$}
  \item A hoodie costs $\pounds 32$ and its price is increased by
        $25\%$. \ans{$\pounds 40$}
        \work{$25\%$ is $\pounds 8$.\\
              $\pounds 32 + \pounds 8 = \pounds 40$}
\end{enumerate}
\end{multicols}

\stepcounter{enumi}\textbf{Q\arabic{enumi}.}~A caf\'e sells $240$ drinks

\vspace{4pt}
How many cold drinks are sold? \ans{$108$}
\work{$35\%+20\%=55\%$, so $45\%$ are cold drinks.\\

%----------------------------------------------------------------
\item A shop raises all of its prices by $20\%$. A month later it takes
$20\%$ off every price in a sale. A customer says:
\vspace{3pt}\par\noindent\begin{minipage}{\linewidth}\centering
\fbox{\begin{minipage}{0.72\linewidth}\small\itshape
``The sale takes off exactly what was added on, so the prices are back
to what they were before.''
\end{minipage}}
\end{minipage}\vspace{3pt}
\vspace{-2pt}
Is the customer right? Use an example price to explain your answer.
\work{No. Take a price of $\pounds 100$.\\
      After the rise: $10\%$ is $\pounds 10$, so $20\%$ is $\pounds 20$,
      giving $\pounds 120$.\\
      In the sale: $10\%$ of $\pounds 120$ is $\pounds 12$, so $20\%$ is
      $\pounds 24$, giving $\pounds 96$.\\
      The price ends up $\pounds 4$ lower, because the $20\%$ taken off
      is $20\%$ of the \emph{larger} price.}

%----------------------------------------------------------------
\item Two shops sell the same headphones, normally $\pounds 80$.
\vspace{3pt}\par\noindent\begin{minipage}{\linewidth}\centering
\fbox{\begin{minipage}[t]{0.40\linewidth}\small\centering
\textbf{Shop A}\par ``$30\%$ off everything!''
\end{minipage}}\hspace{6mm}
\fbox{\begin{minipage}[t]{0.40\linewidth}\small\centering
\textbf{Shop B}\par ``A quarter off, then $\pounds 8$ off!''
\end{minipage}}
\end{minipage}\vspace{3pt}
\vspace{-2pt}
\begin{enumerate}
  \item Which shop is cheaper, and by how much?
        \work{Shop A: $10\%$ is $\pounds 8$, so $30\%$ is $\pounds 24$,
              giving $\pounds 80-\pounds 24=\pounds 56$.\\
              Shop B: a quarter off is $\pounds 20$ off, giving
              $\pounds 60$, then $\pounds 60-\pounds 8=\pounds 52$.\\
              Shop B is cheaper, by $\pounds 4$.}
  \item Design a third offer that beats both shops but still charges
        more than half of the normal price. Explain why it works.
        \work{Many answers. For example ``$40\%$ off'': $10\%$ is
              $\pounds 8$, so $40\%$ is $\pounds 32$, giving
              $\pounds 48$.\\
              That is cheaper than Shop B's $\pounds 52$, and half of
              $\pounds 80$ is $\pounds 40$, so $\pounds 48$ is still
              more than half price.\\
              Any final price between $\pounds 40$ and $\pounds 52$
              works.}
  \item How much would the headphones have to cost for the two shops to charge the same price?
        \work{Shop A takes off $30\%$. Shop B takes off $25\%$ and then
              $\pounds 8$, so Shop A takes off an extra $5\%$ and Shop B
              takes off an extra $\pounds 8$.\\
              The two are the same when $5\%$ is $\pounds 8$, so $10\%$ is
              $\pounds 16$ and $100\%$ is $\pounds 160$.\\
              Check: Shop A gives $\pounds 160-\pounds 48=\pounds 112$;
              Shop B gives $\pounds 160-\pounds 40=\pounds 120$, then
              $\pounds 120-\pounds 8=\pounds 112$.}
\end{enumerate}

\end{enumerate}

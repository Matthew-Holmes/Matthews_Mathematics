\section*{Vector Representations}
\[
\mathbf{u}=\overrightarrow{AB}=\begin{pmatrix}3\\-1\end{pmatrix},\quad
\mathbf{v}=\overrightarrow{PQ}=\begin{pmatrix}5\\3\end{pmatrix},\quad
\mathbf{w}=\overrightarrow{CD}=\begin{pmatrix}4\\3\end{pmatrix},\quad
\mathbf{t}=\overrightarrow{XY}=\begin{pmatrix}3\\3\end{pmatrix}.
\]
The corresponding points are
\[
A=(0,0),\ B=(3,-1),\quad P=(2,3),\ Q=(7,6),\quad C=(-3,2),\ D=(1,5),\quad X=(5,0),\ Y=(8,3).
\]

\section*{Adding Vectors}
\begin{enumerate}
    \item A valid path expression is \(\boxed{\mathbf{g}}\).
    \item Simplified, it is \(\boxed{\mathbf{g}}\).
    \item As a column vector,
    \[
    \boxed{\mathbf{g}=\begin{pmatrix}1\\3\end{pmatrix}}.
    \]
    The resultant is the total displacement from \(R\) to \(S\), equal to the vector \(\mathbf{g}\).
\end{enumerate}

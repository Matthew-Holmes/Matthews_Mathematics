\topic{1. Adding terms}

\method{To add like terms, add their coefficients and keep the variable.
For example, \(3a+2a=(3+2)a=5a\).}

\worked{\(3a+2a=a+a+a+a+a=5a\).}

\textbf{Gap-fill solutions.}
\solitem{\(4b+3b=(4+3)b=7b\).}
\solitem{\(2x+5x=(2+5)x=7x\).}

\textbf{Exercise solutions.}
\begin{solutions}
\solitem{\(5a+2a=7a\)}
\solitem{\(3b+4b=7b\)}
\solitem{\(6x+3x=9x\)}
\solitem{\(2y+8y=10y\)}
\solitem{\(4m+4m=8m\)}
\solitem{\(7p+5p=12p\)}
\solitem{\(a+4a=5a\)}
\solitem{\(9t+t=10t\)}
\solitem{\(c+c=2c\)}
\solitem{\(10n+6n=16n\)}
\solitem{\(2a+3a+4a=(2+3+4)a=9a\)}
\solitem{\(5k+k+6k=12k\)}
\solitem{\(x+x+x+x=4x\)}
\solitem{\(12y+15y=27y\)}
\solitem{\(3d+2d+d+4d=10d\)}
\solitem{\(20z+13z+7z=40z\)}
\end{solutions}

\topic{2. Subtracting terms}

\method{Subtract coefficients in the same way as ordinary numbers, keeping the

\worked{\(8a+3a-5a=11a-5a=6a\).}

\solitem{\(6b+4b-3b=10b-3b=7b\).}

\topic{3. Collecting like terms}

\method{Like terms have exactly the same variables. Collect each set of like
terms separately; unlike terms cannot be combined.}

\worked{\(5a+3b+2a+4b=(5a+2a)+(3b+4b)=7a+7b\).}

\textbf{Gap-fill solutions.}
\solitem{\(4x+2y+3x+5y=(4x+3x)+(2y+5y)=7x+7y\).}
\solitem{\(2m+6n+m+3n=(2m+m)+(6n+3n)=3m+9n\).}

\textbf{Exercise solutions.}
\begin{solutions}
\solitem{\(3a+2b+4a=7a+2b\)}
\solitem{\(5x+2y+3y=5x+5y\)}
\solitem{\(4p+3p+2q=7p+2q\)}
\solitem{\(2x+3x+4y=5x+4y\)}
\solitem{\(2a+3b+4a+b=6a+4b\)}
\solitem{\(6p+q+2p+5q=8p+6q\)}
\solitem{\(3m+4n+3m+4n=6m+8n\)}
\solitem{\(4a+5+3a=7a+5\)}
\solitem{\(x+4+3x+2=4x+6\)}
\solitem{\(6+3t+t=4t+6\)}
\solitem{\(7+2m+5+m=3m+12\)}
\solitem{\(4m+n+2+m=5m+n+2\)}
\solitem{\(5y+1+y+8=6y+9\)}
\solitem{\(3a+2b+6\) cannot be simplified.}
\solitem{\(a+b+a+b+a=3a+2b\)}
\solitem{\(4c+3d+c+2d+5c=(4c+c+5c)+(3d+2d)=10c+5d\)}
\solitem{\(2x+3y+4z+x+y=3x+4y+4z\)}
\solitem{\(3ab+4a+2ab=5ab+4a\)}
\solitem{\(6xy+2x+xy+5x=7xy+7x\)}
\solitem{\(2k+9+3h+4+k+5h=(3h+5h)+(2k+k)+(9+4)=8h+3k+13\)}
\end{solutions}

\topic{4. Collecting like terms with subtraction}

\method{The sign in front of a term belongs to that term. Keep each sign
when collecting like terms.}

\worked{\(7a-2b-3a+6b=(7a-3a)+(-2b+6b)=4a+4b\).}

\textbf{Gap-fill solutions.}
\solitem{\(8x-3y-2x+7y=(8x-2x)+(-3y+7y)=6x+4y\).}
\solitem{\(5m+9n-m-4n=(5m-m)+(9n-4n)=4m+5n\).}

\textbf{Exercise solutions.}
\begin{solutions}
\solitem{\(6a+2b-4a=2a+2b\)}
\solitem{\(5x+7y-3y=5x+4y\)}
\solitem{\(8m-2m+3n=6m+3n\)}
\solitem{\(9a-4a+2b=5a+2b\)}
\solitem{\(9p+4q-2p-q=7p+3q\)}

\solitem{\(4a-3b-a+5b=3a+2b\)}
\solitem{\(8p-3q-5p+7q=3p+4q\)}
\solitem{\(10x+7-6x-3=4x+4\)}
\solitem{\(-2y+5x+6y=5x+4y\)}
\solitem{\(12+5k-4-2k=3k+8\)}
\solitem{\(11k-3-4k+5=7k+2\)}
\solitem{\(7c-5d-2c+9d=5c+4d\)}
\solitem{\(3c+5d-2c+4d-c=(3c-2c-c)+9d=9d\)}

\end{solutions}

\topic{5. Negative coefficients}

\method{If the number being subtracted is larger than the number we start with,
the coefficient is negative. For example, \(3-7=-4\).}

\worked{\(3a-5b-7a+2b=(3a-7a)+(-5b+2b)=-4a-3b\).}

\textbf{Gap-fill solutions.}
\solitem{\(4m-3n-9m+n=(4m-9m)+(-3n+n)=-5m-2n\).}
\solitem{\(p+2q-6p-8q=(p-6p)+(2q-8q)=-5p-6q\).}

\topic{12. Multiplying and collecting like terms}

\method{Multiply first, then collect like terms. Write products with the number
first and the variables in alphabetical order.}

\worked{\(2a\times 3a+4a^2=6a^2+4a^2=10a^2\).}

\textbf{Gap-fill solutions.}
\solitem{\(3x\times 2y+5xy=6xy+5xy=11xy\).}
\solitem{\(4m\times m-m^2=4m^2-m^2=3m^2\).}

\textbf{Missing-term solutions.}
\begin{solutions}
\solitem{\(3a\times 4b=12ab\), so the missing term is \(4b\).}
\solitem{\(5y+3y=8y\), so the missing term is \(3y\).}
\solitem{\(5x^3\times 2x^2=10x^5\), so the missing term is \(5x^3\).}
\solitem{\(4p+3q-2p=2p+3q\), so the missing term is \(3q\).}
\solitem{\(-2x\times -3y=6xy\), so the missing term is \(-2x\).}
\solitem{\(5a^2+(-3a^2)=2a^2\), so the missing term is \(-3a^2\).}
\solitem{\(2m^3\times 4m^4n=8m^7n\), so the missing term is \(4m^4n\).}
\solitem{\(2c+5d+4c+(-6c)=5d\), so the missing term is \(-6c\).}
\end{solutions}

\vspace{8pt}

\solitem{\textbf{Sam's mistake.} \(2a\) and \(3b\) are not like terms, so
\(2a+3b\) cannot be simplified to a single term. Also \(5ab\) means
\(5\times a\times b\), which is a product, not a sum. For example, when
\(a=2\) and \(b=1\), \(2a+3b=4+3=7\), but \(5ab=10\).}

\vspace{6pt}

\solitem{\textbf{Priya's statement.} Priya is not right. \(a^2\) and \(a^3\)
are not like terms, so \(a^2+a^3\) cannot be simplified to \(a^5\).
The multiplication \(a^2\times a^3\) equals \(a^5\). For example, when
\(a=2\), \(a^2+a^3=4+8=12\), but \(a^5=32\).}

\vspace{6pt}

\solitem{\textbf{Tom's mistake.} Tom multiplied the coefficients but did not
multiply the variables:
\[
  3a\times 4a=3\times 4\times a\times a=12a^2.
\]
The correct answer is \(12a^2\), not \(12a\).}

\vspace{6pt}

\solitem{\textbf{Rectangle.} The length is \(5x\) and the width is \(2y\).}
\solitem{(a) Perimeter: \(5x+2y+5x+2y=10x+4y\).}
\solitem{(b) Area: \(5x\times 2y=10xy\).}

\vspace{6pt}

\solitem{\textbf{Three possible pairs of side lengths for area \(12a^2\).}
For example:
\[
  3a \text{ and } 4a,\qquad 2a \text{ and } 6a,\qquad a \text{ and } 12a.
\]
Each pair multiplies to \(12a^2\).}

\vspace{6pt}

\solitem{\textbf{An expression with four terms that simplifies to \(5x-2y\).}
For example:
\[
  7x+y-2x-3y=(7x-2x)+(y-3y)=5x-2y.
\]}

\vspace{6pt}

\solitem{\textbf{Three multiplications that simplify to \(6a^2b\).}
For example:
\[
  2a\times 3ab=6a^2b,\qquad
  6a\times ab=6a^2b,\qquad
  3a^2\times 2b=6a^2b.
\]}

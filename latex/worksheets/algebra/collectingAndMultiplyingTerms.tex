\begin{center}
  {\LARGE\bfseries Collecting and Multiplying Terms}
\end{center}
\vspace{-6pt}

\begin{keyidea}
A \textbf{term} is a number, a variable, or numbers and variables
multiplied together. The \textbf{coefficient} of a term is the number that
multiplies the variables in it.

\vspace{3pt}
To \textbf{simplify} an expression, we write it in a shorter way that is
equal to it for every value of the variables. Follow the worked examples in
this booklet to learn how to simplify expressions.
\end{keyidea}

\vspace{6pt}
\topic{1. Adding terms}

\begin{wex}
\begin{working}
Simplify $3a+2a$.

$3a$ means $a+a+a$.

$2a$ means $a+a$.

$3a+2a=a+a+a+a+a$

So $3a+2a=5a$.
\end{working}
\begin{wexnote}
$3a$ means $3\times a$, which is $a+a+a$. The coefficient of $3a$ is $3$.

In $3a+2a$, the variable $a$ is added $3+2=5$ times. So to add the terms,
we add their coefficients.

A variable on its own has a coefficient of $1$: $b$ means $1b$. So
$b+2b=3b$. We write $b$, not $1b$.
\end{wexnote}
\end{wex}

\topic{2. Subtracting terms}

\begin{wex}
\begin{working}
Simplify $8a+3a-5a$.

We work from left to right.

$8a+3a=11a$

$11a-5a=6a$

So $8a+3a-5a=6a$.
\end{working}
\begin{wexnote}
To subtract terms with the same variable, we subtract their coefficients:
$11-5=6$.

When there are more than two terms, we work from left to right, one term
at a time.

When the coefficient is $1$, we write $a$, not $1a$. When the coefficients
cancel out, the answer is $0$: $4a-4a=0$.
\end{wexnote}
\end{wex}

\topic{3. Collecting like terms}

\begin{wex}
\begin{working}
Simplify $5a+3b+2a+4b$.

The like terms are $5a$ and $2a$, and $3b$ and $4b$.

$5a+2a=7a$

$3b+4b=7b$

So $5a+3b+2a+4b=7a+7b$.
\end{working}
\begin{wexnote}
\textbf{Like terms} have exactly the same variables. $5a$ and $2a$ are like
terms, and $3b$ and $4b$ are like terms.

We collect each set of like terms into a single term.

$7a$ and $7b$ are not like terms, so $7a+7b$ cannot be simplified.

Constants are like terms with each other: $4+2=6$.
\end{wexnote}
\end{wex}

\instruction{Fill in the gaps.}

$4x+3x=$ \gap

\end{scaffold}\hfill

\topic{4. Collecting like terms with subtraction}

\begin{wex}
\begin{working}
Simplify $7a-2b-3a+6b$.

The like terms are $+7a$ and $-3a$, and $-2b$ and $+6b$.

$7a-3a=4a$

$-2b+6b=4b$

So $7a-2b-3a+6b=4a+4b$.
\end{working}
\begin{wexnote}
The sign in front of a term belongs to the term. The terms of
$7a-2b-3a+6b$ are $+7a$, $-2b$, $-3a$ and $+6b$. The first term has no
sign in front of it, so it is positive.

We keep each term's sign when we collect the like terms.

$-2b+6b$ is equal to $6b-2b$, which is $4b$.
\end{wexnote}
\end{wex}

\topic{5. Negative coefficients}

\begin{wex}
\begin{working}
Simplify $3a-5b-7a+2b$.

The like terms are $+3a$ and $-7a$, and $-5b$ and $+2b$.

$3a-7a=-4a$

$-5b+2b=-3b$

So $3a-5b-7a+2b=-4a-3b$.
\end{working}
\begin{wexnote}
When the number we subtract is bigger than the number we start with, the
coefficient is negative. To work out $3-7$, we start at $3$ on the number
line and move $7$ to the left.

\vspace{2pt}
\begin{tikzpicture}[x=0.42cm]
  \draw[-{Stealth[length=5pt]}] (-8.6,0) -- (8.6,0);
  \foreach \n in {-8,...,8} \draw (\n,0.08) -- (\n,-0.08);
  \foreach \n in {-8,-4,0,3,8}
    \node[below,font=\footnotesize,inner sep=2pt] at (\n,-0.08) {$\n$};
  \draw[-{Stealth[length=5pt]},thick,blue!50!black]
    (3,0.15) to[bend right=40] node[above,font=\footnotesize] {move $7$ left} (-4,0.15);
\end{tikzpicture}

So $3-7=-4$, and in the same way $-5+2=-3$.
\end{wexnote}
\end{wex}

\topic{6. Multiplying terms}

\begin{wex}
\begin{working}
Simplify $3a\times 4b$.

$3a\times 4b=3\times 4\times a\times b$

The numbers: $3\times 4=12$.

The variables: $a\times b=ab$.

So $3a\times 4b=12ab$.
\end{working}
\begin{wexnote}
$3a$ means $3\times a$, and $4b$ means $4\times b$.

We can multiply in any order, so we multiply the numbers together and the
variables together.

We write the number first, then the variables in alphabetical order, with
no $\times$ signs: $b\times a$ is written $ab$.
\end{wexnote}
\end{wex}

\topic{7. Multiplying with negative numbers}

\begin{wex}
\begin{working}
Simplify $-2x\times -4y$.

$-2x\times -4y=-2\times -4\times x\times y$

The numbers: $-2\times -4=8$.

The variables: $x\times y=xy$.

So $-2x\times -4y=8xy$.
\end{working}
\begin{wexnote}
When we multiply two numbers with the same sign, the answer is positive.
When the signs are different, the answer is negative.

\vspace{2pt}
\begin{tabular}{@{}l@{\qquad}l@{}}
$3\times 4=12$     & $-3\times 4=-12$\\[2pt]
$-3\times -4=12$   & $3\times -4=-12$
\end{tabular}

$-a$ means $-1\times a$, so the coefficient of $-a$ is $-1$.
\end{wexnote}
\end{wex}

$-3a\times 5b=-3\times 5\times a\times b$

\topic{8. Squares and cubes}

\begin{wex}
\begin{working}
Simplify $3a\times 2a$.

$3a\times 2a=3\times 2\times a\times a$

The numbers: $3\times 2=6$.

The variables: $a\times a=a^2$.

So $3a\times 2a=6a^2$.
\end{working}
\begin{wexnote}
$a\times a$ is written $a^2$, which we say as ``$a$ squared''.
$a\times a\times a$ is written $a^3$, which we say as ``$a$ cubed''.

The small number is called the \textbf{index}. It tells us how many times
the variable is multiplied: $a^4=a\times a\times a\times a$.

$a^2b$ means $a\times a\times b$.
\end{wexnote}
\end{wex}

\topic{9. Multiplying powers}

\begin{wex}
\begin{working}
Simplify $2a^3\times 4a^2$.

$2a^3\times 4a^2=2\times 4\times a^3\times a^2$

The numbers: $2\times 4=8$.

The variables:

$a^3\times a^2=a^{3+2}=a^5$

So $2a^3\times 4a^2=8a^5$.
\end{working}
\begin{wexnote}
$a^3\times a^2=\underbrace{a\times a\times a}_{a^3}\times
  \underbrace{a\times a}_{a^2}=a^5$

When we multiply powers of the same variable, we add the indices:
$3+2=5$.

A variable with no index has an index of $1$: $a=a^1$. So
$a^4\times a=a^{4+1}=a^5$.
\end{wexnote}
\end{wex}

\topic{10. Multiplying terms with more than one variable}

\begin{wex}
\begin{working}
Simplify $3a^2b\times 2ab^3$.

$3a^2b\times 2ab^3$\\
\hspace*{1em}$=3\times 2\times a^2\times a\times b\times b^3$

The numbers: $3\times 2=6$.

The powers of $a$: $a^2\times a=a^3$.

The powers of $b$: $b\times b^3=b^4$.

So $3a^2b\times 2ab^3=6a^3b^4$.
\end{working}
\begin{wexnote}
$a^2b$ means $a^2\times b$, and $ab^3$ means $a\times b^3$.

We multiply the numbers, and then we multiply the powers of each variable
separately. We add the indices of $a$, and then we add the indices of
$b$.

We do not add the indices of different variables: $a^3\times b^4$ is
written $a^3b^4$.
\end{wexnote}
\end{wex}

\topic{11. Collecting like terms with powers}

\begin{wex}
\begin{working}
Simplify $5a^2+3a-2a^2+4a$.

The like terms are $+5a^2$ and $-2a^2$, and $+3a$ and $+4a$.

$5a^2-2a^2=3a^2$

$3a+4a=7a$

So $5a^2+3a-2a^2+4a=3a^2+7a$.
\end{working}
\begin{wexnote}
Like terms have exactly the same variables, with the same indices.
$5a^2$ and $-2a^2$ are like terms. $5a^2$ and $3a$ are not like terms,
because $a^2=a\times a$ is not the same as $a$.

When we collect like terms, we add the coefficients and the index stays
the same: $a^2+a^2=2a^2$.

$a^2b$ and $ab^2$ are not like terms.
\end{wexnote}
\end{wex}

\topic{12. Multiplying and collecting like terms}

\begin{wex}
\begin{working}
Simplify $2a\times 3a+4a^2$.

We multiply first:

$2a\times 3a=6a^2$

So $2a\times 3a+4a^2=6a^2+4a^2$.

We collect the like terms:

$6a^2+4a^2=10a^2$

So $2a\times 3a+4a^2=10a^2$.
\end{working}
\begin{wexnote}
We multiply before we add or subtract, in the same way as
$2\times 3+4=6+4=10$.

After we multiply, we write each term with the number first and the
variables in alphabetical order. Then we can see which terms are like
terms: $3y\times 2x=6xy$, which is a like term with $xy$.
\end{wexnote}
\end{wex}

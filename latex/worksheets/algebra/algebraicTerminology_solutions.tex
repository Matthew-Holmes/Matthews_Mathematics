\begin{center}
  {\LARGE\bfseries Algebraic Terminology}\\[2pt]
  {\large Answer Key}
\end{center}

\begin{description}
\setlength{\itemsep}{2pt}
\setlength{\parsep}{0pt}

\item[\textbf{Q1.}] Coefficient \(=4\); constant \(=+9\).
\item[\textbf{Q2.}] Coefficient \(=2\); variable \(=a\); term \(=+6b\); constant \(=+5\).
\item[\textbf{Q3.}] \(3\) terms; variables \(x,y,z\); coefficient of \(xy\) is \(5\); constant \(7\).
\item[\textbf{Q4.}] Coefficient of \(n\) is \(9\); coefficient of \(p\) is \(3\); \(4\) is the constant.
\item[\textbf{Q5.}] \(2p+7\) (or \(7+2p\)).
\item[\textbf{Q6.}] Example: \(4a+3b+8\). Many other answers are possible.
\item[\textbf{Q7.}] Coefficient \(=7\); variable \(=y\); constant \(=-2\); expression \(=7y-2\).
\item[\textbf{Q8.}] Coefficient \(=4\); variable \(=a\); term \(=+9b\); constant \(=-6\).
\item[\textbf{Q9.}] Coefficient \(=3\); variables \(=p,q\); constant \(=-8\).
\item[\textbf{Q10.}] \(3\) terms; variables \(x,y,z\); coefficient of \(xy\) is \(4\); constant \(-5\).
\item[\textbf{Q11.}] \(4\) terms; coefficient of \(m\) is \(6\); coefficient of \(n\) is \(-1\); \(8\) is the constant.
\item[\textbf{Q12.}] \(2\) terms; variables \(a,b\); coefficient of \(ab\) is \(-3\); constant \(7\).
\item[\textbf{Q13.}] Coefficient of \(b\) is \(1\); coefficient of \(c\) is \(-4\); constant is \(-9\).
\item[\textbf{Q14.}] \(3p-5q+4\).
\item[\textbf{Q15.}] Example: \(5a+2b+ab-1\). Many other answers are possible.
\item[\textbf{Q16.}] Constant \(=8\); coefficient \(=-3\); variable \(=x\); expression \(=8-3x\).
\item[\textbf{Q17.}] Coefficient \(=2\); constant \(=-7\); variable \(=b\); term \(=+0.5b\).
\item[\textbf{Q18.}] Coefficient \(=-4\); variable \(=m\); term \(=-4mn\); constant \(=+1.5\).
\item[\textbf{Q19.}] \(4\) terms; coefficient of \(a\) is \(0.5\); coefficient of \(b\) is \(-1\); constant \(2.5\).
\item[\textbf{Q20.}] \(2\) terms; variable \(k\); constant \(9\); coefficient of \(k^2\) is \(-0.5\).
\item[\textbf{Q21.}] \(3\) terms; coefficient of \(xy\) is \(2\); \(-5\) is the coefficient of \(x\); no constant.
\item[\textbf{Q22.}] \(5\) terms; coefficient of \(x^2\) is \(1\); coefficient of \(xy\) is \(1\); constant \(-3\).
\item[\textbf{Q23.}] \(-x+0.5y+2z-3\).
\item[\textbf{Q24.}] Example: \(-a-2b-3c-ab+0.5\). Many other answers are possible.

\end{description}

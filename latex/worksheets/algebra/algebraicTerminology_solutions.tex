\begin{center}
  {\Large\bfseries Algebraic Terminology: Answer Key}
\end{center}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*]
  \item \(7y-2\): coefficient \answer{\(7\)}; variable \answer{\(y\)}; constant \answer{\(-2\)}; expression \answer{\(7y-2\)}.
  \item \(4a+9b-6\): terms \answer{\(4a\), \(+9b\), \(-6\)}; coefficient of \(a\) \answer{\(4\)}; constant \answer{\(-6\)}; the term \(9b\) has coefficient \(9\) and variable \(b\).
  \item \(4xy+2z-5\): \answer{\(3\)} terms (\(4xy\), \(2z\), \(-5\)); variables \answer{\(x,y,z\)}; coefficient of \(xy\) \answer{\(4\)}; constant \answer{\(-5\)}.
  \item \(6m-n+8\): terms \answer{\(6m\), \(-n\), \(8\)}; coefficient of \(m\) \answer{\(6\)}; coefficient of \(n\) \answer{\(-1\)}; \(8\) is the \answer{constant}.
  \item \answer{\(3p-5q+4\)} (or any reordering).
  \item One possible answer: \answer{\(5a+2b+ab-1\)} (also e.g.\ \(3x+4y-xy-1\)).
\end{enumerate}

\section*{Part 3: Matching Sequences to Their \(n\)th Terms}

\textbf{Q11.}
\begin{itemize}
\item[(a)] \(6,12,18,24,\ldots\to 6n\)
\item \(0.4,0.8,1.2,1.6,\ldots\to 0.4n\)
\item \(-4,-8,-12,-16,\ldots\to -4n\)
\item \(\frac14,\frac12,\frac34,1,\ldots\to 0.25n\) or \(\frac14 n\)
\item \(20\%,40\%,60\%,80\%,\ldots\to 20n\%\)
\item \(-0.5,-1,-1.5,-2,\ldots\to -0.5n\)

\item[(b)] \(n+6\): \(7,8,9,10\); \(0.5n\): \(0.5,1,1.5,2\)
\end{itemize}

\textbf{Q28.} Sample answers:
\begin{itemize}
\item Always: Every term of \(6n\) is a multiple of \(3\).
\item Sometimes: A term of \(4n\) is greater than \(10\). True for \(n=3\), false for \(n=1\).
\item Never: A term of \(5n\) is negative. Since \(n=1,2,3,\ldots\), every term is positive.
\end{itemize}

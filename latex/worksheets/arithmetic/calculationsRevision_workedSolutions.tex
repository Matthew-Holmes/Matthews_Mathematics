\section*{Section 1: Addition and Subtraction}

\begin{multicols}{2}
\begin{enumerate}
    \item $45 + 376 =$
    
    \textbf{Solution:}
    \begin{align*}
        45 + 376 &= 45 + 300 + 76 \\
        &= 345 + 76 \\
        &= 345 + 50 + 26 \\
        &= 395 + 26 \\
        &= 421
    \end{align*}
    \textit{Alternatively:} Line up vertically:
    \begin{align*}
        &\ \ 45 \\
        +&\ 376 \\
        \hline
        &\ 421
    \end{align*}
    
    \item $328.5 + 19.47 =$
    
    \textbf{Solution:} Line up decimal points:
    \begin{align*}
        &\ 328.50 \quad \text{(add trailing zero for clarity)} \\
        +&\ \ 19.47 \\
        \hline
        &\ 347.97
    \end{align*}
    \[328.5 + 19.47 = 347.97\]
    
    \item $7.63 + (-2.8) =$
    
    \textbf{Solution:} Adding a negative is the same as subtraction:
    \begin{align*}
        7.63 + (-2.8) &= 7.63 - 2.8 \\
        &= 7.63 - 2.80 \quad \text{(line up decimals)} \\
        &= 4.83
    \end{align*}
    
    \item $-12 + 7 =$
    
    \textbf{Solution:} Think of a number line: start at -12, move 7 to the right:
    \[-12 + 7 = -5\]
    \textit{Alternatively:} $12 - 7 = 5$, but since -12 has larger magnitude, answer is negative: $-5$.
    
    \item $56.3 - 18.72 =$
    
    \textbf{Solution:} Line up decimals, borrow as needed:
    \begin{align*}
        &\ \ 56.30 \quad \text{(add trailing zero)} \\
        -&\ \ 18.72 \\
        \hline
        &\ \ 37.58
    \end{align*}
    \[56.3 - 18.72 = 37.58\]
    
    \item $-4 - (-9) =$
    
    \textbf{Solution:} Subtracting a negative is the same as adding:
    \begin{align*}
        -4 - (-9) &= -4 + 9 \\
        &= 5
    \end{align*}
\end{enumerate}
\end{multicols}

\textbf{Solutions:}

\begin{enumerate}
    \item $24 \times 6 = 144$
    
    \textbf{Working:} $20 \times 6 = 120$, $4 \times 6 = 24$, total $120 + 24 = 144$.
    
    \item $-2 \times -3 \times -2 = -12$
    
    \textbf{Working:} Count negatives: three negatives (odd number) gives negative answer.
    \[(-2 \times -3) \times -2 = (6) \times -2 = -12\]
    
    \item $0.25 \times 0.4 = 0.1$
    
    \textbf{Working:} Method 1 - Counting decimal places:
    \[25 \times 4 = 100\]
    2 decimal places in 0.25 + 1 decimal place in 0.4 = 3 decimal places total.
    So answer is $0.100 = 0.1$.
    
    \item $1.2 \times 0.06 = 0.072$
    
    \textbf{Working:} $12 \times 6 = 72$, 1 decimal place + 2 decimal places = 3 decimal places: $0.072$.
    
    \item $3.4 \times 1.2 = 4.08$ (as shown in example)
    
    \item $5.6 \times (-0.3) = -1.68$
    
    \textbf{Working:} $56 \times 3 = 168$, 1 decimal place + 1 decimal place = 2 decimal places: $1.68$.
    One negative: answer is negative: $-1.68$.
    
    \item $12.5 \times 0.08 = 1.0$
    
    \textbf{Working:} $125 \times 8 = 1000$, 1 decimal place + 2 decimal places = 3 decimal places: $1.000 = 1.0$.
    
    \item $0.09 \times 0.7 = 0.063$
    
    \textbf{Working:} $9 \times 7 = 63$, 2 decimal places + 1 decimal place = 3 decimal places: $0.063$.
    
    \item $-3 \times 4 = -12$
    
    \textbf{Working:} One negative: answer negative. $3 \times 4 = 12$, so $-12$.
    
    \item $0.0034 \times 68.7 = 0.23358$
    
    \textbf{Working:} $34 \times 687 = 23358$ (calculate: $34 \times 600 = 20400$, $34 \times 87 = 2958$, total $23358$).
    4 decimal places + 1 decimal place = 5 decimal places: $0.23358$.
    
    \item $-4.2 \times 0.5 \times -0.5 = 1.05$
    
    \textbf{Working:} Two negatives: answer positive.
    First: $4.2 \times 0.5 = 2.1$ ($42 \times 5 = 210$, 1+1=2 decimal places: $2.10$).
    Then: $2.1 \times 0.5 = 1.05$ ($21 \times 5 = 105$, 1+1=2 decimal places: $1.05$).
    
    \item $-0.6 \times -0.3 \times -2 = -0.36$
    
    \textbf{Working:} Three negatives: answer negative.
    First: $0.6 \times 0.3 = 0.18$ ($6 \times 3 = 18$, 1+1=2 decimal places).
    Then: $0.18 \times 2 = 0.36$.
    With negative sign: $-0.36$.
\end{enumerate}

\section*{Section 3: Division}

\begin{enumerate}
    \item $84 \div 7 = 12$
    
    \textbf{Working:} $7 \times 12 = 84$, or bus stop:
    \begin{align*}
        7\overline{)84} &\quad 7 \text{ into } 8 = 1 \text{ remainder } 1 \\
        &\quad \text{Bring down } 4, 7 \text{ into } 14 = 2 \\
        &\quad \text{Answer: } 12
    \end{align*}
    
    \item $156 \div 12 = 13$
    
    \textbf{Working:} Chunking: $156 \div 4 = 39$, then $39 \div 3 = 13$.
    Or: $12 \times 13 = 156$.
    
    \item $48 \div (-6) = -8$
    
    \textbf{Working:} One negative: answer negative. $48 \div 6 = 8$, so $-8$.
    
    \item $4.8 \div 0.4 = 12$
    
    \textbf{Working:} Multiply both by 10: $48 \div 4 = 12$.
    
    \item $-9 \div -3 = 3$
    
    \textbf{Working:} Two negatives: answer positive. $9 \div 3 = 3$.
    
    \item $8280 \div 24 = 345$
    
    \textbf{Working:} Bus stop method:
    \begin{align*}
        24\overline{)8280} &\quad 24 \text{ into } 82 = 3 \text{ (}24\times3=72\text{)} \\
        &\quad 82-72=10, \text{ bring down } 8 = 108 \\
        &\quad 24 \text{ into } 108 = 4 \text{ (}24\times4=96\text{)} \\
        &\quad 108-96=12, \text{ bring down } 0 = 120 \\
        &\quad 24 \text{ into } 120 = 5 \text{ (}24\times5=120\text{)} \\
        &\quad \text{Answer: } 345
    \end{align*}
    
    \item $3.6 \div 0.6 = 6$
    
    \textbf{Working:} Multiply both by 10: $36 \div 6 = 6$.
    
    \item $0.84 \div 0.07 = 12$
    
    \textbf{Working:} Multiply both by 100: $84 \div 7 = 12$.
    
    \item $7.2 \div 0.6 = 12$
    
    \textbf{Working:} Multiply both by 10: $72 \div 6 = 12$.
    
    \item $-12.6 \div 0.3 = -42$
    
    \textbf{Working:} One negative: answer negative.
    Multiply both by 10: $126 \div 3 = 42$, so $-42$.
    
    \item $0.56 \div 0.08 = 7$
    
    \textbf{Working:} Multiply both by 100: $56 \div 8 = 7$.
    
    \item $6.4 \div (-0.2) = -32$
    
    \textbf{Working:} One negative: answer negative.
    Multiply both by 10: $64 \div 2 = 32$, so $-32$.
    
    \item $-1.8 \div 0.3 = -6$
    
    \textbf{Working:} One negative: answer negative.
    Multiply both by 10: $18 \div 3 = 6$, so $-6$.
    
    \item $-0.45 \div 0.05 = -9$
    
    \textbf{Working:} One negative: answer negative.
    Multiply both by 100: $45 \div 5 = 9$, so $-9$.
\end{enumerate}

\section*{Section 4: Mixed Practice}

\begin{enumerate}
    \item $12.5 + (-3.4) = 9.1$
    
    \textbf{Working:} $12.5 - 3.4 = 9.1$.
    
    \item $-3 \times 4 \times -2 = 24$
    
    \textbf{Working:} Two negatives: answer positive.
    $3 \times 4 = 12$, $12 \times 2 = 24$.
    
    \item $24 \div (-6) + 5 \times 2 = 6$
    
    \textbf{Working:} BODMAS: division and multiplication first.
    $24 \div (-6) = -4$, $5 \times 2 = 10$.
    Then: $-4 + 10 = 6$.
    
    \item $6.3 \div 0.3 = 21$
    
    \textbf{Working:} Multiply both by 10: $63 \div 3 = 21$.
    
    \item $(-2.4 \times 3.5) \div -0.7 = 12$
    
    \textbf{Working:} First: $-2.4 \times 3.5 = -8.4$ (one negative).
    Then: $-8.4 \div -0.7 = 12$ (two negatives: positive).
    Check: $8.4 \div 0.7 = 84 \div 7 = 12$.
    
    \item $-15 + 28.4 \div 4 = -8.9$
    
    \textbf{Working:} Division first: $28.4 \div 4 = 7.1$.
    Then: $-15 + 7.1 = -7.9$.
    
    \item $7.25 - 3.68 = 3.57$
    
    \textbf{Working:} Line up decimals:
    \begin{align*}
        &\ 7.25 \\
        -&\ 3.68 \\
        \hline
        &\ 3.57
    \end{align*}
    
    \item $-8 \times (5 - 7) = 16$
    
    \textbf{Working:} Brackets first: $5 - 7 = -2$.
    Then: $-8 \times -2 = 16$ (two negatives: positive).
    
    \item $9.6 \div 1.2 + (-3) \times 2 = 2$
    
    \textbf{Working:} Division and multiplication first:
    $9.6 \div 1.2 = 8$ (multiply both by 10: $96 \div 12 = 8$).
    $(-3) \times 2 = -6$.
    Then: $8 + (-6) = 8 - 6 = 2$.
    
    \item $-36 \div (-9) = 4$
    
    \textbf{Working:} Two negatives: positive. $36 \div 9 = 4$.
\end{enumerate}

\section*{Extension Word Problems}

\begin{enumerate}
    \item The temperature was $-3^\circ$C and dropped by $5^\circ$C overnight. What is the new temperature?
    
    \textbf{Solution:} Dropping means subtracting:
    \[-3 - 5 = -8\]
    New temperature: $-8^\circ$C.
    
    \item A scuba diver descends $2.5$ m per minute for 4 minutes, then ascends $3$ m. What is the diver's final depth?
    
    \textbf{Solution:} Descend: $2.5 \times 4 = 10$ m down (negative if surface is 0).
    Ascend: $3$ m up (positive).
    Starting at surface (0 m):
    \[0 - 10 + 3 = -7\]
    Final depth: $7$ m below surface.
    
    \item A submarine starts at sea level and travels vertically $-4.2$ km, then another $-3.8$ km. How far is it from the surface now? What is the submarine's altitude?
    
    \textbf{Solution:} Both movements are downward (negative):
    \[0 + (-4.2) + (-3.8) = -8.0\]
    Distance from surface: $8.0$ km (absolute value).
    Altitude: $-8.0$ km (8 km below sea level).
    
    \item Some numbers produce recurring decimals when used as divisors. Find the decimal result of 1 divided by each of the first 7 prime numbers. Some results terminate while others recur. Can you explain why?
    
    \textbf{Solution:} First 7 primes: 2, 3, 5, 7, 11, 13, 17.
    \begin{align*}
        1 \div 2 &= 0.5 \quad \text{(terminates)} \\
        1 \div 3 &= 0.\overline{3} \quad \text{(repeats)} \\
        1 \div 5 &= 0.2 \quad \text{(terminates)} \\
        1 \div 7 &= 0.\overline{142857} \quad \text{(repeats)} \\
        1 \div 11 &= 0.\overline{09} \quad \text{(repeats)} \\
        1 \div 13 &= 0.\overline{076923} \quad \text{(repeats)} \\
        1 \div 17 &= 0.\overline{0588235294117647} \quad \text{(repeats)}
    \end{align*}
\end{enumerate}
